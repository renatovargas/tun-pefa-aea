1 الغرض

تحسب هذه الصفحة حسابات انبعاثات الهواء (AEA) لتونس الخاصة بغازات الاحتباس الحراري، من 2015 إلى 2022 (Eurostat, 2015; United Nations وآخرون, 2014). وكما في حسابات PEFA، تقوم بذلك بطريقتين.

الطبقة المرجعية تستدعي الدالة I2A من أداة صندوق النقد الدولي (International Monetary Fund, 2026; Pegoue, 2026)، دون تعديلها، عبر الوحدة scripts/outils_fmi.py1. والمعطيات هي معطيات الصيغة الواردة في الورقة Output_Source_Activity من المصنف.

الطبقة الصريحة تعيد كل خطوة من I2A بمكتبة polars، في هذه الصفحة. شيفرة هذه الطبقة مقتبسة من I2A (أشيل بيغوي، صندوق النقد الدولي)؛ وتحيل كل خطوة إلى الأسطر المقابلة في I2A.py. وهذه الطبقة هي التي تنتج الحسابات.

تتلقى الطبقتان الجرد نفسه (الصفحة 20)، وتجري مقارنتهما على ثلاث مجموعات من المعطيات. وهي المصفوفة الافتراضية دون بيانات منظمة التعاون والتنمية في الميدان الاقتصادي، والمصفوفة الافتراضية مع هذه البيانات، والمصفوفة التونسية (الصفحة 21) مع هذه البيانات.

DESCRIPTIONS = {ligne[1]: ligne[2] for ligne in plages["CRT"][1:]}
ENTETE_INVENTAIRE = ["Info", "Data_Source", "IPCC_annex", "CountryRegion", "REF_AREA", "CountryName",

def plage_inventaire(inventaire: pl.DataFrame, annees: list[int]) -> list[list]:
    """Plage Inventory_2 : une ligne par gaz et code de l'outil, une colonne par année (kt éq. CO2)."""
    large = (
        inventaire.filter(pl.col("annee").is_in(annees))
        .group_by("gaz", "code_outil", "annee")
        .agg(pl.col("kt_eq_co2").sum())
        .pivot("annee", index=["gaz", "code_outil"], values="kt_eq_co2")
        .sort("gaz", "code_outil")
    )
    lignes = [ENTETE_INVENTAIRE + list(annees)]
    for r in large.iter_rows(named=True):
        lignes.append(
            ["Most detailed Component", "UNFCCC", "Non-Annex_I", "Africa", "TUN", "Tunisia", r["gaz"], r["code_outil"],
             DESCRIPTIONS.get(r["code_outil"]), "fossil"]
            + [float(r.get(str(a)) or 0.0) for a in annees]
        )
    return lignes

3 الطبقة الصريحة

3.1 المدخلات

تحتفظ الأداة بالمكونات الأكثر تفصيلًا ذات المنشأ الأحفوري، ثم تضع الجرد في صيغة طويلة. ومع بيانات منظمة التعاون والتنمية في الميدان الاقتصادي، تضيف سلسلتين بأطنان CO2. في النقل الجوي، هي انبعاثات الناقلين المقيمين. تُحفظ الرحلات المحلية (RES_DOM_IN) على حدة؛ وتُجمع الرحلات المحلية خارج البلد والرحلات الدولية تحت RES_ABROAD. وفي النقل البحري، هي انبعاثات جميع السفن التي يشغّلها مقيمون (ALL_VESSELS). وتتلقى السلسلتان رمز الطيران المحلي ورمز الملاحة المحلية، المقروءين في العمود Comment من جدول فئات CRT.

يُعاد ضبط المصفوفة لكي يساوي مجموعها 1 لكل غاز ولكل فئة. ويصلح السطر الذي لا غاز فيه للغازات الأربعة في الأداة. ويبقى السطر الذي كل معاملاته أصفار مساويًا للصفر.

GAZ_OUTIL = ["CH4", "CO2", "N2O", "F-gases"]
CODE_AIR = next(l[1] for l in plages["CRT"][1:] if l[11] == "Domestic Aviation")
CODE_MER = next(l[1] for l in plages["CRT"][1:] if l[11] == "Domestic Navigation")
RESIDENTS_ETRANGER = ["RES_DOM_OUT", "RES_INT_FROM", "RES_INT_TO", "RES_INT_OUT"]


def plage_en_table(plage: list[list]) -> pl.DataFrame:
    return pl.DataFrame(plage[1:], schema=[str(c) for c in plage[0]], orient="row", infer_schema_length=None)


def inventaire_long(plage: list[list]) -> pl.DataFrame:
    """Lignes les plus détaillées et fossiles, en format long (lignes 87 à 118 et 237 à 261 de I2A.py)."""
    table = plage_en_table(plage)
    annees = [c for c in table.columns if c not in ENTETE_INVENTAIRE]
    return (
        table.filter(pl.col("Info") == "Most detailed Component", (pl.col("FossilType") == "fossil") | pl.col("FossilType").is_null())
        .unpivot(index=["POLLUTANT", "Source_Code"], on=annees, variable_name="annee", value_name="valeur")
        .with_columns(pl.col("annee").cast(pl.Int16), pl.col("valeur").cast(pl.Float64), source=pl.lit("INV"))
    )


def oecd_long(oecd: dict, annees: list[int]) -> pl.DataFrame:
    """Émissions de l'OCDE en kt (lignes 144 à 232) : résidents à l'étranger (aérien) et tous navires (maritime)."""
    air = plage_en_table(oecd["OECD_Air"]).select(
        "POLLUTANT", "TIME_PERIOD", "OBS_VALUE",
        source=pl.col("EMISSIONS_SOURCE").replace(dict.fromkeys(RESIDENTS_ETRANGER, "RES_ABROAD")),
        Source_Code=pl.lit(CODE_AIR),
    )
    mer = plage_en_table(oecd["OECD_Sea"]).select(
        "POLLUTANT", "TIME_PERIOD", "OBS_VALUE", source=pl.col("VESSEL"), Source_Code=pl.lit(CODE_MER)
    )
    return (
        pl.concat([air, mer])
        .with_columns(annee=pl.col("TIME_PERIOD").cast(pl.Int16), valeur=pl.col("OBS_VALUE").cast(pl.Float64).fill_null(0) / 1000)
        .filter(pl.col("annee").is_in(annees))
        .select("POLLUTANT", "Source_Code", "annee", "valeur", "source")
    )


def matrice_longue(plage: list[list]) -> pl.DataFrame:
    """Facteurs normalisés à 1 par gaz et code (lignes 271 à 293) ; un gaz vide vaut pour tous les gaz."""
    table = plage_en_table(plage).drop("Level", strict=False).with_columns(pl.col("POLLUTANT").cast(pl.String))
    colonnes = [c for c in table.columns if c not in ("POLLUTANT", "Source_Code", "Source_Description", "Check")]
    longue = (
        table.unpivot(index=["POLLUTANT", "Source_Code"], on=colonnes, variable_name="Activity", value_name="facteur")
        .with_columns(pl.col("facteur").cast(pl.Float64).fill_null(0))
        .with_columns(somme=pl.col("facteur").sum().over("POLLUTANT", "Source_Code"))
        .with_columns(facteur=pl.when(pl.col("somme") == 0).then(0.0).otherwise(pl.col("facteur") / pl.col("somme")))
    )
    vides = longue.filter(pl.col("POLLUTANT").is_null() | (pl.col("POLLUTANT") == ""))
    if vides.height:
        longue = pl.concat([
            longue.filter(pl.col("POLLUTANT").is_not_null() & (pl.col("POLLUTANT") != "")),
            *[vides.with_columns(POLLUTANT=pl.lit(g)) for g in GAZ_OUTIL],
        ])
    return longue.select("POLLUTANT", "Source_Code", "Activity", "facteur")

3.2 تعديلات الإقامة

تمر بيانات المنظمة عبر المصفوفة بمعاملات رمزها، مثل الجرد. ثم تقارنها الأداة، نشاطًا نشاطًا، بانبعاثات الجرد للرمز نفسه.

  • الرحلات المحلية. إذا أصدر المقيمون أكثر مما في الجرد، يُضاف الفارق إلى النشاط، ويُضاف بالإشارة المعاكسة إلى التعديلات الأخرى (OTHER). وإذا أصدروا أقل، يُطرح الفارق من النشاط ويُسند إلى غير المقيمين في الإقليم (NRT-AIR). ولا تعطي المنظمة سوى CO2. لذلك يذهب CH4 وN2O من الطيران المحلي كاملين إلى غير المقيمين.
  • رحلات المقيمين في الخارج. تُضاف هذه الانبعاثات إلى النشاط وتظهر أيضًا في بند الربط NRA-AIR.
  • السفن. إذا أصدرت سفن المقيمين أكثر مما في الجرد، يُضاف الفارق إلى النشاط. وتنوي الأداة تسجيله أيضًا في بند الربط NRA-WATER، لكن اختبارها (السطر 425) يبحث عن هذا الرمز بين أرقام أسطر جدول الأنشطة لا بين رموزه. لذلك يبقى بند الربط فارغًا. وإذا أصدرت أقل، يذهب الفارق إلى غير المقيمين (NRT-WATER).
def ajustements_residence(attribue: pl.DataFrame, codes_activite: list[str]) -> pl.DataFrame:
    """Ajustements de résidence de l'outil (lignes 395 à 514), par gaz, code, activité et année.

    INV est l'inventaire attribué ; les autres colonnes sont les données de l'OCDE attribuées
    avec les mêmes facteurs.
    """
    large = (
        attribue.filter(pl.col("Source_Code").is_in([CODE_AIR, CODE_MER]))
        .pivot("source", index=["POLLUTANT", "Source_Code", "Activity", "annee"], values="valeur", aggregate_function="sum")
    )
    sources = [c for c in large.columns if c not in ("POLLUTANT", "Source_Code", "Activity", "annee")]
    large = large.with_columns(pl.col(sources).fill_null(0))
    if "INV" not in large.columns:
        large = large.with_columns(INV=pl.lit(0.0))
    lignes = []

    def ajouter(table: pl.DataFrame, source: pl.Expr, activite: pl.Expr, valeur: pl.Expr) -> None:
        lignes.append(table.select("POLLUTANT", "annee", Source_Code=source, Activity=activite, valeur=valeur))

    if "ALL_VESSELS" in sources:
        mer = large.filter(pl.col("Source_Code") == CODE_MER).with_columns(d=pl.col("ALL_VESSELS") - pl.col("INV"))
        positif, negatif = mer.filter(pl.col("d") > 0), mer.filter(pl.col("d") < 0)
        # Écart positif : ajouté à l'activité, sans élément de passage (ligne 425)
        ajouter(positif, pl.lit("ALL_VESSELS"), pl.col("Activity"), pl.col("d"))
        # Écart négatif : retiré de l'activité et attribué aux non-résidents sur le territoire
        ajouter(negatif, pl.col("Source_Code"), pl.col("Activity"), pl.col("d"))
        if "NRT-WATER" in codes_activite:
            ajouter(negatif, pl.col("Source_Code"), pl.lit("NRT-WATER"), -pl.col("d"))
    if "RES_DOM_IN" in sources:
        air = large.filter(pl.col("Source_Code") == CODE_AIR).with_columns(d=pl.col("RES_DOM_IN") - pl.col("INV"))
        positif, negatif = air.filter(pl.col("d") > 0), air.filter(pl.col("d") < 0)
        ajouter(positif, pl.lit("RES_DOM_IN"), pl.col("Activity"), pl.col("d"))
        if "OTHER" in codes_activite:
            ajouter(positif, pl.lit("RES_DOM_IN"), pl.lit("OTHER"), -pl.col("d"))
        ajouter(negatif, pl.col("Source_Code"), pl.col("Activity"), pl.col("d"))
        if "NRT-AIR" in codes_activite:
            ajouter(negatif, pl.col("Source_Code"), pl.lit("NRT-AIR"), -pl.col("d"))
    if "RES_ABROAD" in sources:
        air = large.filter(pl.col("Source_Code") == CODE_AIR)
        ajouter(air, pl.lit("RES_ABROAD"), pl.col("Activity"), pl.col("RES_ABROAD"))
        if "NRA-AIR" in codes_activite:
            ajouter(air, pl.lit("RES_ABROAD"), pl.lit("NRA-AIR"), pl.col("RES_ABROAD"))
    return pl.concat(lignes) if lignes else attribue.head(0).select("POLLUTANT", "annee", "Source_Code", "Activity", "valeur")

3.3 التجميع

تربط الدالة الكاملة الخطوات بعضها ببعض. وتضيف مجموعين. يجمع AEA-TOTAL الأنشطة الواقعة قبل هذا السطر في جدول الأنشطة، أي الفروع والأسر. ويجمع INV-TOTAL الأسطر الواقعة بين AEA-TOTAL وINV-TOTAL، وهي مجموع الحسابات وبنود الربط وLULUCF. وأخيرًا، تضيف الدالة مجموع غازات الاحتباس الحراري (GHG).

def i2a_explicite(plage_inv: list[list], plage_matrice: list[list], plage_activite: list[list], oecd: dict | None = None) -> pl.DataFrame:
    """Reprise de la fonction I2A en polars, résultat en kt d'équivalent CO2."""
    inv = inventaire_long(plage_inv)
    annees = sorted(inv["annee"].unique().to_list())
    entrees = pl.concat([inv, oecd_long(oecd, annees)]) if oecd else inv
    entrees = entrees.group_by("POLLUTANT", "Source_Code", "annee", "source").agg(pl.col("valeur").sum())

    # Attribution aux activités (lignes 313 à 335) ; un code sans facteur ne reçoit rien
    attribue = (
        entrees.join(matrice_longue(plage_matrice), on=["POLLUTANT", "Source_Code"], how="inner")
        .with_columns(valeur=pl.col("valeur") * pl.col("facteur"))
        .filter(pl.col("valeur") != 0)
    )
    codes_activite = [l[0] for l in plage_activite[1:]]
    if oecd and attribue["source"].n_unique() > 2:
        resultat = pl.concat([
            attribue.filter(pl.col("source") == "INV").select("POLLUTANT", "annee", "Source_Code", "Activity", "valeur"),
            ajustements_residence(attribue, codes_activite),
        ])
    else:
        resultat = attribue.select("POLLUTANT", "annee", "Source_Code", "Activity", "valeur")
    resultat = resultat.filter(pl.col("valeur") != 0)

    # Totaux (lignes 528 à 557) : les activités avant AEA-TOTAL, puis de AEA-TOTAL à INV-TOTAL
    i_aea, i_inv = codes_activite.index("AEA-TOTAL"), codes_activite.index("INV-TOTAL")
    aea = resultat.filter(pl.col("Activity").is_in(codes_activite[:i_aea])).with_columns(
        Source_Code=pl.lit("Total AEA"), Activity=pl.lit("AEA-TOTAL")
    )
    resultat = pl.concat([resultat, aea])
    total_inventaire = resultat.filter(pl.col("Activity").is_in(codes_activite[i_aea:i_inv])).with_columns(
        Source_Code=pl.lit("Total Inventories"), Activity=pl.lit("INV-TOTAL")
    )
    resultat = pl.concat([resultat, total_inventaire]).group_by(CLES).agg(pl.col("valeur").sum())

    # Total des gaz à effet de serre (lignes 563 à 571)
    ges = resultat.group_by("Source_Code", "Activity", "annee").agg(pl.col("valeur").sum()).with_columns(POLLUTANT=pl.lit("GHG"))
    return pl.concat([resultat, ges.select(resultat.columns)])


def comparer(outil: pl.DataFrame, explicite: pl.DataFrame) -> pl.DataFrame:
    a = outil.group_by(CLES).agg(pl.col("valeur").sum().alias("outil"))
    b = explicite.group_by(CLES).agg(pl.col("valeur").sum().alias("explicite"))
    return (
        a.join(b, on=CLES, how="full", coalesce=True)
        .with_columns(pl.col("outil", "explicite").fill_null(0))
        .with_columns(ecart=pl.col("outil") - pl.col("explicite"))
    )

4 التحقق بالمصفوفة الافتراضية

resultats_defaut = {}
for nom, options in [("sans OCDE", None), ("avec OCDE", oecd)]:
    choix = of.OECD_UTILISER if options else of.OECD_NE_PAS_UTILISER
    outil = of.i2a(inventaire_outil, plages["Conversion_Matrix"], plages["CRT"], plages["NFR"], plages["Activity"], oecd, choix)
    explicite = i2a_explicite(inventaire_outil, plages["Conversion_Matrix"], plages["Activity"], options)
    ecarts = comparer(outil, explicite)
    assert ecarts["ecart"].abs().max() < TOLERANCE_KT, f"Les deux couches diffèrent ({nom})"
    resultats_defaut[nom] = explicite
    print(f"Matrice par défaut, {nom} : {ecarts.height} cellules, écart maximal {ecarts['ecart'].abs().max():.2e} kt")

5 حسابات تونس

5.1 نطاقات المصفوفة التونسية

تُمرَّر المصفوفة التونسية إلى الأداة بصيغة ورقتها Conversion_Matrix، سنة بسنة، لأن مفاتيح التوزيع تتغير كل سنة. وأعمدتها هي فروع جداول الموارد والاستخدامات الأربعة والعشرون والأسر وLULUCF. ويبقى عمود الغازات فارغًا، فتصلح المصفوفة لجميع الغازات. ويبدأ جدول الأنشطة بالفروع والأسر، ثم يأخذ أسطر الأداة ابتداءً من AEA-TOTAL. وبذلك تبقى مجاميع الأداة معرّفة بالطريقة نفسها.

ACTIVITES_TUNISIE = [c for c in activites["code"] if c not in ("RDM", "ACC", "ENV", "ECART")]
COLONNES_MATRICE = ACTIVITES_TUNISIE + ["LULUCF"]


def plage_matrice_tunisie(annee: int) -> list[list]:
    """Plage Conversion_Matrix de la matrice tunisienne d'une année (gaz vide : tous les gaz)."""
    large = (
        matrice_tunisie.filter(pl.col("annee") == annee)
        .pivot("activite", index="code_outil", values="facteur", aggregate_function="sum")
        .fill_null(0)
        .sort("code_outil")
    )
    lignes = [["POLLUTANT", "Level", "Source_Code", "Source_Description", "Check"] + COLONNES_MATRICE]
    for r in large.iter_rows(named=True):
        lignes.append([None, None, r["code_outil"], DESCRIPTIONS.get(r["code_outil"]), 100.0]
                      + [100.0 * float(r.get(c) or 0.0) for c in COLONNES_MATRICE])

def plage_activite_tunisie() -> list[list]:

## الغرض

تحسب هذه الصفحة حسابات انبعاثات الهواء (AEA) لتونس الخاصة بغازات الاحتباس الحراري، من 2015 إلى 2022 [@eurostat2015aea; @seea2014]. وكما في حسابات PEFA، تقوم بذلك بطريقتين.

**الطبقة المرجعية** تستدعي الدالة `I2A` من أداة صندوق النقد الدولي [@pegoue2026i2a; @imf2026i2a]، دون تعديلها، عبر الوحدة `scripts/outils_fmi.py`[^outils]. والمعطيات هي معطيات الصيغة الواردة في الورقة `Output_Source_Activity` من المصنف.

**الطبقة الصريحة** تعيد كل خطوة من `I2A` بمكتبة polars، في هذه الصفحة. شيفرة هذه الطبقة مقتبسة من `I2A` (أشيل بيغوي، صندوق النقد الدولي)؛ وتحيل كل خطوة إلى الأسطر المقابلة في `I2A.py`. وهذه الطبقة هي التي تنتج الحسابات.

تتلقى الطبقتان الجرد نفسه (الصفحة 20)، وتجري مقارنتهما على ثلاث مجموعات من المعطيات. وهي المصفوفة الافتراضية دون بيانات منظمة التعاون والتنمية في الميدان الاقتصادي، والمصفوفة الافتراضية مع هذه البيانات، والمصفوفة التونسية (الصفحة 21) مع هذه البيانات.

[^outils]: الشيفرة المصدرية: <https://github.com/renatovargas/tun-pefa-aea/blob/main/scripts/outils_fmi.py>. الأداة: <https://github.com/renatovargas/tun-pefa-aea/tree/main/outils-fmi/i2a>.

```{python}
GAZ_OUTIL = ["CH4", "CO2", "N2O", "F-gases"]

def plage_en_table(plage: list[list]) -> pl.DataFrame:
    return pl.DataFrame(plage[1:], schema=[str(c) for c in plage[0]], orient="row", infer_schema_length=None)


def inventaire_long(plage: list[list]) -> pl.DataFrame:

    table = plage_en_table(plage)

        .with_columns(pl.col("annee").cast(pl.Int16), pl.col("valeur").cast(pl.Float64), source=pl.lit("INV"))
    )


def oecd_long(oecd: dict, annees: list[int]) -> pl.DataFrame:
    """Émissions de l'OCDE en kt (lignes 144 à 232) : résidents à l'étranger (aérien) et tous navires (maritime)."""
    air = plage_en_table(oecd["OECD_Air"]).select(
        "POLLUTANT", "TIME_PERIOD", "OBS_VALUE",

    )

        "POLLUTANT", "TIME_PERIOD", "OBS_VALUE", source=pl.col("VESSEL"), Source_Code=pl.lit(CODE_MER)
    )
    return (
        pl.concat([air, mer])
        .with_columns(annee=pl.col("TIME_PERIOD").cast(pl.Int16), valeur=pl.col("OBS_VALUE").cast(pl.Float64).fill_null(0) / 1000)
        .filter(pl.col("annee").is_in(annees))
        .select("POLLUTANT", "Source_Code", "annee", "valeur", "source")
    )


def matrice_longue(plage: list[list]) -> pl.DataFrame:
    """Facteurs normalisés à 1 par gaz et code (lignes 271 à 293) ; un gaz vide vaut pour tous les gaz."""
    table = plage_en_table(plage).drop("Level", strict=False).with_columns(pl.col("POLLUTANT").cast(pl.String))
    colonnes = [c for c in table.columns if c not in ("POLLUTANT", "Source_Code", "Source_Description", "Check")]
    longue = (
        table.unpivot(index=["POLLUTANT", "Source_Code"], on=colonnes, variable_name="Activity", value_name="facteur")
        .with_columns(pl.col("facteur").cast(pl.Float64).fill_null(0))
        .with_columns(somme=pl.col("facteur").sum().over("POLLUTANT", "Source_Code"))
        .with_columns(facteur=pl.when(pl.col("somme") == 0).then(0.0).otherwise(pl.col("facteur") / pl.col("somme")))
    )
    vides = longue.filter(pl.col("POLLUTANT").is_null() | (pl.col("POLLUTANT") == ""))
    if vides.height:
        longue = pl.concat([
            longue.filter(pl.col("POLLUTANT").is_not_null() & (pl.col("POLLUTANT") != "")),
            *[vides.with_columns(POLLUTANT=pl.lit(g)) for g in GAZ_OUTIL],
        ])
    return longue.select("POLLUTANT", "Source_Code", "Activity", "facteur")
```

### تعديلات الإقامة

تمر بيانات المنظمة عبر المصفوفة بمعاملات رمزها، مثل الجرد. ثم تقارنها الأداة، نشاطًا نشاطًا، بانبعاثات الجرد للرمز نفسه.

- **الرحلات المحلية.** إذا أصدر المقيمون أكثر مما في الجرد، يُضاف الفارق إلى النشاط، ويُضاف بالإشارة المعاكسة إلى التعديلات الأخرى (`OTHER`). وإذا أصدروا أقل، يُطرح الفارق من النشاط ويُسند إلى غير المقيمين في الإقليم (`NRT-AIR`). ولا تعطي المنظمة سوى CO2. لذلك يذهب CH4 وN2O من الطيران المحلي كاملين إلى غير المقيمين.
- **رحلات المقيمين في الخارج.** تُضاف هذه الانبعاثات إلى النشاط وتظهر أيضًا في بند الربط `NRA-AIR`.
- **السفن.** إذا أصدرت سفن المقيمين أكثر مما في الجرد، يُضاف الفارق إلى النشاط. وتنوي الأداة تسجيله أيضًا في بند الربط `NRA-WATER`، لكن اختبارها (السطر 425) يبحث عن هذا الرمز بين أرقام أسطر جدول الأنشطة لا بين رموزه. لذلك يبقى بند الربط فارغًا. وإذا أصدرت أقل، يذهب الفارق إلى غير المقيمين (`NRT-WATER`).

```{python}
#| label: explicite-residence
def ajustements_residence(attribue: pl.DataFrame, codes_activite: list[str]) -> pl.DataFrame:
    """Ajustements de résidence de l'outil (lignes 395 à 514), par gaz, code, activité et année.

    INV est l'inventaire attribué ; les autres colonnes sont les données de l'OCDE attribuées
    avec les mêmes facteurs.
    """
    large = (
        attribue.filter(pl.col("Source_Code").is_in([CODE_AIR, CODE_MER]))
        .pivot("source", index=["POLLUTANT", "Source_Code", "Activity", "annee"], values="valeur", aggregate_function="sum")
    )
    sources = [c for c in large.columns if c not in ("POLLUTANT", "Source_Code", "Activity", "annee")]
    large = large.with_columns(pl.col(sources).fill_null(0))
    if "INV" not in large.columns:
        large = large.with_columns(INV=pl.lit(0.0))
    lignes = []

    def ajouter(table: pl.DataFrame, source: pl.Expr, activite: pl.Expr, valeur: pl.Expr) -> None:
        lignes.append(table.select("POLLUTANT", "annee", Source_Code=source, Activity=activite, valeur=valeur))

    if "ALL_VESSELS" in sources:
        mer = large.filter(pl.col("Source_Code") == CODE_MER).with_columns(d=pl.col("ALL_VESSELS") - pl.col("INV"))
        positif, negatif = mer.filter(pl.col("d") > 0), mer.filter(pl.col("d") < 0)
        # Écart positif : ajouté à l'activité, sans élément de passage (ligne 425)
        ajouter(positif, pl.lit("ALL_VESSELS"), pl.col("Activity"), pl.col("d"))
        # Écart négatif : retiré de l'activité et attribué aux non-résidents sur le territoire
        ajouter(negatif, pl.col("Source_Code"), pl.col("Activity"), pl.col("d"))
        if "NRT-WATER" in codes_activite:
            ajouter(negatif, pl.col("Source_Code"), pl.lit("NRT-WATER"), -pl.col("d"))
    if "RES_DOM_IN" in sources:
        air = large.filter(pl.col("Source_Code") == CODE_AIR).with_columns(d=pl.col("RES_DOM_IN") - pl.col("INV"))
        positif, negatif = air.filter(pl.col("d") > 0), air.filter(pl.col("d") < 0)
        ajouter(positif, pl.lit("RES_DOM_IN"), pl.col("Activity"), pl.col("d"))
        if "OTHER" in codes_activite:
            ajouter(positif, pl.lit("RES_DOM_IN"), pl.lit("OTHER"), -pl.col("d"))
        ajouter(negatif, pl.col("Source_Code"), pl.col("Activity"), pl.col("d"))
        if "NRT-AIR" in codes_activite:
            ajouter(negatif, pl.col("Source_Code"), pl.lit("NRT-AIR"), -pl.col("d"))
    if "RES_ABROAD" in sources:
        air = large.filter(pl.col("Source_Code") == CODE_AIR)
        ajouter(air, pl.lit("RES_ABROAD"), pl.col("Activity"), pl.col("RES_ABROAD"))
        if "NRA-AIR" in codes_activite:
            ajouter(air, pl.lit("RES_ABROAD"), pl.lit("NRA-AIR"), pl.col("RES_ABROAD"))
    return pl.concat(lignes) if lignes else attribue.head(0).select("POLLUTANT", "annee", "Source_Code", "Activity", "valeur")
```

### التجميع

تربط الدالة الكاملة الخطوات بعضها ببعض. وتضيف مجموعين. يجمع `AEA-TOTAL` الأنشطة الواقعة قبل هذا السطر في جدول الأنشطة، أي الفروع والأسر. ويجمع `INV-TOTAL` الأسطر الواقعة بين `AEA-TOTAL` و`INV-TOTAL`، وهي مجموع الحسابات وبنود الربط وLULUCF. وأخيرًا، تضيف الدالة مجموع غازات الاحتباس الحراري (`GHG`).

```{python}
#| label: explicite-assemblage
def i2a_explicite(plage_inv: list[list], plage_matrice: list[list], plage_activite: list[list], oecd: dict | None = None) -> pl.DataFrame:
    """Reprise de la fonction I2A en polars, résultat en kt d'équivalent CO2."""
    inv = inventaire_long(plage_inv)
    annees = sorted(inv["annee"].unique().to_list())
    entrees = pl.concat([inv, oecd_long(oecd, annees)]) if oecd else inv
    entrees = entrees.group_by("POLLUTANT", "Source_Code", "annee", "source").agg(pl.col("valeur").sum())

    # Attribution aux activités (lignes 313 à 335) ; un code sans facteur ne reçoit rien
    attribue = (
        entrees.join(matrice_longue(plage_matrice), on=["POLLUTANT", "Source_Code"], how="inner")
        .with_columns(valeur=pl.col("valeur") * pl.col("facteur"))
        .filter(pl.col("valeur") != 0)
    )
    codes_activite = [l[0] for l in plage_activite[1:]]
    if oecd and attribue["source"].n_unique() > 2:
        resultat = pl.concat([
            attribue.filter(pl.col("source") == "INV").select("POLLUTANT", "annee", "Source_Code", "Activity", "valeur"),
            ajustements_residence(attribue, codes_activite),
        ])
    else:
        resultat = attribue.select("POLLUTANT", "annee", "Source_Code", "Activity", "valeur")
    resultat = resultat.filter(pl.col("valeur") != 0)

    # Totaux (lignes 528 à 557) : les activités avant AEA-TOTAL, puis de AEA-TOTAL à INV-TOTAL
    i_aea, i_inv = codes_activite.index("AEA-TOTAL"), codes_activite.index("INV-TOTAL")
    aea = resultat.filter(pl.col("Activity").is_in(codes_activite[:i_aea])).with_columns(
        Source_Code=pl.lit("Total AEA"), Activity=pl.lit("AEA-TOTAL")
    )
    resultat = pl.concat([resultat, aea])
    total_inventaire = resultat.filter(pl.col("Activity").is_in(codes_activite[i_aea:i_inv])).with_columns(
        Source_Code=pl.lit("Total Inventories"), Activity=pl.lit("INV-TOTAL")
    )
    resultat = pl.concat([resultat, total_inventaire]).group_by(CLES).agg(pl.col("valeur").sum())

    # Total des gaz à effet de serre (lignes 563 à 571)
    ges = resultat.group_by("Source_Code", "Activity", "annee").agg(pl.col("valeur").sum()).with_columns(POLLUTANT=pl.lit("GHG"))
    return pl.concat([resultat, ges.select(resultat.columns)])


def comparer(outil: pl.DataFrame, explicite: pl.DataFrame) -> pl.DataFrame:
    a = outil.group_by(CLES).agg(pl.col("valeur").sum().alias("outil"))
    b = explicite.group_by(CLES).agg(pl.col("valeur").sum().alias("explicite"))
    return (
        a.join(b, on=CLES, how="full", coalesce=True)
        .with_columns(pl.col("outil", "explicite").fill_null(0))
        .with_columns(ecart=pl.col("outil") - pl.col("explicite"))
    )
```

## حسابات تونس

### نطاقات المصفوفة التونسية

تُمرَّر المصفوفة التونسية إلى الأداة بصيغة ورقتها `Conversion_Matrix`، سنة بسنة، لأن مفاتيح التوزيع تتغير كل سنة. وأعمدتها هي فروع جداول الموارد والاستخدامات الأربعة والعشرون والأسر وLULUCF. ويبقى عمود الغازات فارغًا، فتصلح المصفوفة لجميع الغازات. ويبدأ جدول الأنشطة بالفروع والأسر، ثم يأخذ أسطر الأداة ابتداءً من `AEA-TOTAL`. وبذلك تبقى مجاميع الأداة معرّفة بالطريقة نفسها.

```{python}
#| label: tunisie-plages
ACTIVITES_TUNISIE = [c for c in activites["code"] if c not in ("RDM", "ACC", "ENV", "ECART")]
COLONNES_MATRICE = ACTIVITES_TUNISIE + ["LULUCF"]


def plage_matrice_tunisie(annee: int) -> list[list]:
    """Plage Conversion_Matrix de la matrice tunisienne d'une année (gaz vide : tous les gaz)."""
    large = (
        matrice_tunisie.filter(pl.col("annee") == annee)
        .pivot("activite", index="code_outil", values="facteur", aggregate_function="sum")
        .fill_null(0)
        .sort("code_outil")
    )
    lignes = [["POLLUTANT", "Level", "Source_Code", "Source_Description", "Check"] + COLONNES_MATRICE]
    for r in large.iter_rows(named=True):
        lignes.append([None, None, r["code_outil"], DESCRIPTIONS.get(r["code_outil"]), 100.0]
                      + [100.0 * float(r.get(c) or 0.0) for c in COLONNES_MATRICE])
    return lignes


def plage_activite_tunisie() -> list[list]:
    """Plage Activity : branches du TRE et ménages, puis les lignes de l'outil à partir de AEA-TOTAL."""
    entete, *defaut = plages["Activity"]
    debut = next(i for i, l in enumerate(defaut) if l[0] == "AEA-TOTAL")
    libelles = dict(activites.select("code", "fr").iter_rows())
    propres = [[c, libelles[c]] + [None] * (len(entete) - 2) for c in ACTIVITES_TUNISIE]
    return [entete] + propres + defaut[debut:]


activite_tunisie = plage_activite_tunisie()
[l[:2] for l in activite_tunisie[-17:]]
```
