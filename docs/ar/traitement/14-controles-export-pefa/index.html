1 الغرض

تتحقق هذه الصفحة من حسابات PEFA المحسوبة في الصفحة السابقة، وتقارنها بجداول الموارد والاستخدامات للتأكد من معقوليتها، وتحسب كثافة الطاقة حسب الفرع، وتصدّر كل ذلك في مصنف إكسل بالفرنسية والإنجليزية والعربية.

from pathlib import Path

import polars as pl
import xlsxwriter

NOM = "comptes_pefa"

ANNEES = sorted(pefa["annee"].unique().to_list())

# Codes d'activité de l'outil et codes du site

2 توازن الموارد والاستخدامات

في جدول PEFA، تكون كل فئة من التدفقات متوازنة. فالموارد الطبيعية التي توفرها البيئة تساوي تلك التي تستعملها الفروع، والبقايا المنتجة تساوي البقايا التي تمتصها البيئة أو تتراكم. وفي المنتجات الطاقوية، تساوي الموارد الاستخدامات بصرف النظر عن فارق التقريب في الميزان.

par_categorie = (
    pefa.group_by("annee", "niveau1")
    .agg(
        pl.col("tj").filter(pl.col("quadrant") == "Supply").sum().alias("ressources_tj"),
        pl.col("tj").filter(pl.col("quadrant") == "Use").sum().alias("emplois_tj"),
    )
    .with_columns(ecart_ktep=((pl.col("ressources_tj") - pl.col("emplois_tj")) / KTEP_EN_TJ).round(2))
)
par_categorie.pivot("niveau1", index="annee", values="ecart_ktep").sort("annee")
residus = par_categorie.filter(pl.col("niveau1") != "Energy products:")
assert residus["ecart_ktep"].abs().max() < 1e-6, "Ressources naturelles ou résidus déséquilibrés"

في المنتجات الطاقوية، يجري التحقق منتجًا منتجًا. تسجل الأداة جميع الاستخدامات غير الطاقوية في سطر واحد، مهما كان المنتج. في تونس، لا تخص هذه الاستخدامات إلا المنتجات النفطية، وهو ما يؤكده الميزان. لذلك يُلحق هذا السطر بالنفط لأغراض التحقق.

non_energetique = modele.filter(pl.col("flux_modele").str.starts_with("Non-energy use"))
petroliers = pl.read_csv("data/nomenclatures/bilan_produits.csv").filter(pl.col("parent") == "PP")["code"]
assert non_energetique["code_produit"].is_in(petroliers.implode()).all()

produits_energie = (
    pefa.filter(pl.col("niveau1") == "Energy products:")
    .with_columns(
        produit=pl.when(pl.col("produit") == "End-use of energy products for non-energy purposes").then(pl.lit("Oil")).otherwise("produit")
    )
    .group_by("annee", "produit")
    .agg(
        pl.col("tj").filter(pl.col("quadrant") == "Supply").sum().alias("ressources_tj"),
        pl.col("tj").filter(pl.col("quadrant") == "Use").sum().alias("emplois_tj"),
    )
    .with_columns(ecart_ktep=((pl.col("ressources_tj") - pl.col("emplois_tj")) / KTEP_EN_TJ).round(2))
)
assert produits_energie["ecart_ktep"].abs().max() <= 5, "Écart supérieur à l'arrondi du bilan"
produits_energie.pivot("produit", index="annee", values="ecart_ktep").sort("annee")

تنتج الفوارق المتبقية عن تقريب خلايا الميزان المنشورة إلى ألف طن مكافئ نفط، وقد سبقت الإشارة إليه في الصفحة 10. يجمع النفط عشرة منتجات من الميزان تتراكم فوارق تقريبها؛ لذلك يبلغ هامش التسامح 5 آلاف طن مكافئ نفط.

3 المعقولية بالمقارنة مع جداول الموارد والاستخدامات

يجب على الفرع الذي يستعمل الطاقة أن يشتري، في جداول الموارد والاستخدامات، النفط والغاز الطبيعي (المنتج 06) أو المنتجات النفطية المكررة (19) أو الكهرباء والغاز (35). هذه الجداول بالقيمة، لذلك تنصبّ المقارنة على الإشارة ورتبة الحجم. وتعطي النسبة بينهما سعرًا ضمنيًا متوسطًا، بالدينار لكل جيغاجول، ينبغي أن يبقى في نطاق معقول من فرع إلى آخر.

ANNEE = ANNEES[-1]
SUT = set(activites.filter(~pl.col("code").is_in(PSEUDO + ["MEN"]))["code"])

energie_finale = (
    pefa.filter(
        pl.col("quadrant") == "Use",
        pl.col("niveau1") == "Energy products:",
        pl.col("niveau2") == "End-use of energy products by SIEC class:",
        pl.col("activite").is_in(list(SUT | {"MEN"})),
    )
    .group_by("annee", "activite")
    .agg(pl.col("tj").sum().alias("energie_tj"))
)
achats = (
    tre.filter(pl.col("bloc") == "emplois", pl.col("ligne").is_in(["06", "19", "35"]), ~pl.col("agregat"), pl.col("annee").is_in(ANNEES))
    .with_columns(activite=pl.when(pl.col("colonne") == "P3_MEN").then(pl.lit("MEN")).otherwise(pl.col("colonne")))
    .filter(pl.col("activite").is_in(list(SUT | {"MEN"})))
    .group_by("annee", "activite")
    .agg(pl.col("valeur").sum().alias("achats_mdt"))
)
vraisemblance = (
    energie_finale.join(achats, on=["annee", "activite"], how="full", coalesce=True)
    .with_columns(pl.col("energie_tj", "achats_mdt").fill_null(0))
    .with_columns(prix_tnd_gj=(pl.col("achats_mdt") * 1e6 / (pl.col("energie_tj") * 1e3)).round(1))
    .join(activites.select(pl.col("code").alias("activite"), "fr"), on="activite")
)
vraisemblance.filter(pl.col("annee") == ANNEE).sort("energie_tj", descending=True).with_columns(
    pl.col("energie_tj", "achats_mdt").round(0)
)

للتكرير (19) والكهرباء (35) سعر ضمني مرتفع، لأن مشترياتهما من المنتجات الطاقوية هي في معظمها مدخلات تحويل، وهي مستبعدة هنا من الاستخدام النهائي. أما الفروع التي تستهلك الطاقة دون أن تشتريها في الجداول، أو العكس، فتستحق الفحص؛ ويدرجها الجدول التالي.

vraisemblance.filter(
    ((pl.col("energie_tj") > 0) & (pl.col("achats_mdt") <= 0)) | ((pl.col("energie_tj") <= 0) & (pl.col("achats_mdt") > 0))
).select("annee", "activite", "fr", "energie_tj", "achats_mdt").sort("activite", "annee")

تشير الأسعار الضمنية المرتفعة جدًا أو المنخفضة جدًا إلى توزيع ينبغي مراجعته. تتلقى فروع الخدمات طاقة قطاع الخدمات حسب مشترياتها من الكهرباء والغاز؛ ويعكس سعرها الضمني هذا المفتاح أساسًا. والفروع التي تشتري منتجات نفطية دون أن تتلقى طاقة من الميزان (لمركباتها مثلًا) تتوقف على مفتاح النقل البري. ستُناقش هذه النقاط مع المعهد الوطني للإحصاء خلال البعثة.

4 كثافة الطاقة

تقيس كثافة الطاقة في فرع ما نسبة استخدامه النهائي للطاقة (دون مدخلات التحويل) إلى إنتاجه أو قيمته المضافة، بالتيراجول لكل مليون دينار بالأسعار الجارية. تتبع هذه المؤشرات جداول الموارد والاستخدامات بالأسعار الجارية، لذلك يجمع تطورها بين الكميات والأسعار.

comptes_production = (
    tre.filter(pl.col("bloc") == "valeur_ajoutee", pl.col("ligne").is_in(["P1", "B1G"]))
    .pivot("ligne", index=["annee", "colonne"], values="valeur")
    .rename({"colonne": "activite", "P1": "production_mdt", "B1G": "valeur_ajoutee_mdt"})
)
indicateurs = (
    energie_finale.filter(pl.col("activite") != "MEN")
    .join(comptes_production, on=["annee", "activite"], how="left")
    .with_columns(
        intensite_production=pl.col("energie_tj") / pl.col("production_mdt"),
        intensite_va=pl.col("energie_tj") / pl.col("valeur_ajoutee_mdt"),
    )
    .rename({"energie_tj": "energie_finale_tj"})
    .sort("annee", "activite")
)
(
    indicateurs.filter(pl.col("annee") == ANNEE)
    .join(activites.select(pl.col("code").alias("activite"), "fr"), on="activite")
    .select("activite", "fr", pl.col("energie_finale_tj").round(0), pl.col("intensite_va").round(2))
    .sort("intensite_va", descending=True)
)

5 التصدير

يحتوي المصنف على ورقة ملاحظات، والجدول الطويل لجميع السنوات، وورقة موارد وورقة استخدامات لكل سنة، والمؤشرات. تأتي تسميات الأسطر من data/nomenclatures/pefa_lignes.csv، وتسميات الأنشطة من data/nomenclatures/activites.csv، والعناوين من data/nomenclatures/libelles.csv. وتتجه الأوراق العربية من اليمين إلى اليسار.

ORDRE_ACTIVITES = activites["code"].to_list()
ORDRE_LIGNES = {(t, c): o for t, c, o in lignes.select("type", "cle", "ordre").iter_rows()}


def traduire(langue: str) -> tuple[dict, dict, dict]:
    t = dict(zip(libelles["cle"], libelles[langue]))
    l = dict(zip(lignes["cle"], lignes[langue]))
    a = dict(zip(activites["code"], activites[langue]))
    return t, l, a


def matrice(annee: int, quadrant: str, langue: str) -> pl.DataFrame:
    """Tableau des ressources ou des emplois d'une année : produits en lignes, activités en colonnes."""
    t, l, a = traduire(langue)
    donnees = pefa.filter(pl.col("annee") == annee, pl.col("quadrant") == quadrant)
    presentes = [c for c in ORDRE_ACTIVITES if c in set(donnees["activite"])]
    tableau = (
        donnees.with_columns(
            o1=pl.col("niveau1").replace_strict({c: o for (ty, c), o in ORDRE_LIGNES.items() if ty == "niveau1"}),
            o2=pl.col("niveau2").replace_strict({c: o for (ty, c), o in ORDRE_LIGNES.items() if ty == "niveau2"}, default=0),
            o3=pl.col("produit").replace_strict({c: o for (ty, c), o in ORDRE_LIGNES.items() if ty == "produit"}),
        )
        .pivot("activite", index=["o1", "o2", "o3", "niveau1", "niveau2", "produit"], values="tj", aggregate_function="sum")
        .sort("o1", "o2", "o3")
    )
    tableau = tableau.with_columns(pl.col(presentes).fill_null(0)).with_columns(
        pl.sum_horizontal(presentes).alias("total")
    )
    return tableau.select(
        pl.col("niveau1").replace_strict(l).alias(t["niveau1"]),
        pl.col("niveau2").replace_strict(l, default=None).alias(t["niveau2"]),
        pl.col("produit").replace_strict(l).alias(t["produit"]),
        *[pl.col(c).alias(f"{c} {a[c]}") for c in presentes],
        pl.col("total").alias(t["total"]),
    )


def donnees_longues(langue: str) -> pl.DataFrame:
    t, l, a = traduire(langue)
    return pefa.select(
        pl.col("annee").alias(t["annee"]),
        pl.col("quadrant").replace_strict({"Supply": t["Supply"], "Use": t["Use"]}).alias(t["quadrant"]),
        pl.col("niveau1").replace_strict(l).alias(t["niveau1"]),
        pl.col("niveau2").replace_strict(l, default=None).alias(t["niveau2"]),
        pl.col("produit").replace_strict(l).alias(t["produit"]),
        pl.col("activite").alias(t["code_activite"]),
        pl.col("activite").replace_strict(a).alias(t["activite"]),
        pl.col("tj").alias(t["tj"]),
    )


def indicateurs_traduits(langue: str) -> pl.DataFrame:
    t, l, a = traduire(langue)
    return indicateurs.select(
        pl.col("annee").alias(t["annee"]),
        pl.col("activite").alias(t["code_activite"]),
        pl.col("activite").replace_strict(a).alias(t["activite"]),
        *[pl.col(c).alias(t[c]) for c in ["energie_finale_tj", "production_mdt", "valeur_ajoutee_mdt", "intensite_production", "intensite_va"]],
    )


def exporter(langue: str) -> Path:
    t, _, _ = traduire(langue)
    dossier = PRODUITS / langue
    dossier.mkdir(parents=True, exist_ok=True)
    chemin = dossier / f"{NOM}.xlsx"
    with xlsxwriter.Workbook(chemin) as classeur:
        notes = classeur.add_worksheet(t["feuille_notes"])
        for i, cle in enumerate(["titre_pefa", "unite_tj", "note_source", "note_methode", "note_provisoire", "note_site"]):
            notes.write(i, 0, t[cle])
        feuilles = [(t["feuille_donnees"], donnees_longues(langue), 1)]
        for annee in ANNEES:
            feuilles.append((f"{t['feuille_ressources']} {annee}", matrice(annee, "Supply", langue), 1))
            feuilles.append((f"{t['feuille_emplois']} {annee}", matrice(annee, "Use", langue), 1))
        feuilles.append((t["feuille_indicateurs"], indicateurs_traduits(langue), 3))
        for nom, tableau, decimales in feuilles:
            tableau.write_excel(workbook=classeur, worksheet=nom, autofit=True, float_precision=decimales)
        if langue == "ar":
            for feuille in classeur.worksheets():
                feuille.right_to_left()
    return chemin


for langue in ["fr", "en", "ar"]:
    print(exporter(langue))

```{python}
    .with_columns(
        produit=pl.when(pl.col("produit") == "End-use of energy products for non-energy purposes").then(pl.lit("Oil")).otherwise("produit")
    )
    .group_by("annee", "produit")
    .agg(
        pl.col("tj").filter(pl.col("quadrant") == "Supply").sum().alias("ressources_tj"),
        pl.col("tj").filter(pl.col("quadrant") == "Use").sum().alias("emplois_tj"),
    )
    .with_columns(ecart_ktep=((pl.col("ressources_tj") - pl.col("emplois_tj")) / KTEP_EN_TJ).round(2))
)
assert produits_energie["ecart_ktep"].abs().max() <= 5, "Écart supérieur à l'arrondi du bilan"
produits_energie.pivot("produit", index="annee", values="ecart_ktep").sort("annee")
```

تنتج الفوارق المتبقية عن تقريب خلايا الميزان المنشورة إلى ألف طن مكافئ نفط، وقد سبقت الإشارة إليه في الصفحة 10. يجمع النفط عشرة منتجات من الميزان تتراكم فوارق تقريبها؛ لذلك يبلغ هامش التسامح 5 آلاف طن مكافئ نفط.

## المعقولية بالمقارنة مع جداول الموارد والاستخدامات

يجب على الفرع الذي يستعمل الطاقة أن يشتري، في جداول الموارد والاستخدامات، النفط والغاز الطبيعي (المنتج 06) أو المنتجات النفطية المكررة (19) أو الكهرباء والغاز (35). هذه الجداول بالقيمة، لذلك تنصبّ المقارنة على الإشارة ورتبة الحجم. وتعطي النسبة بينهما سعرًا ضمنيًا متوسطًا، بالدينار لكل جيغاجول، ينبغي أن يبقى في نطاق معقول من فرع إلى آخر.

```{python}
#| label: vraisemblance
ANNEE = ANNEES[-1]
SUT = set(activites.filter(~pl.col("code").is_in(PSEUDO + ["MEN"]))["code"])

energie_finale = (
    pefa.filter(
        pl.col("quadrant") == "Use",
        pl.col("niveau1") == "Energy products:",
        pl.col("niveau2") == "End-use of energy products by SIEC class:",
        pl.col("activite").is_in(list(SUT | {"MEN"})),
    )
    .group_by("annee", "activite")
    .agg(pl.col("tj").sum().alias("energie_tj"))
)
achats = (
    tre.filter(pl.col("bloc") == "emplois", pl.col("ligne").is_in(["06", "19", "35"]), ~pl.col("agregat"), pl.col("annee").is_in(ANNEES))
    .with_columns(activite=pl.when(pl.col("colonne") == "P3_MEN").then(pl.lit("MEN")).otherwise(pl.col("colonne")))
    .filter(pl.col("activite").is_in(list(SUT | {"MEN"})))
    .group_by("annee", "activite")
    .agg(pl.col("valeur").sum().alias("achats_mdt"))
)
vraisemblance = (
    energie_finale.join(achats, on=["annee", "activite"], how="full", coalesce=True)
    .with_columns(pl.col("energie_tj", "achats_mdt").fill_null(0))
    .with_columns(prix_tnd_gj=(pl.col("achats_mdt") * 1e6 / (pl.col("energie_tj") * 1e3)).round(1))
    .join(activites.select(pl.col("code").alias("activite"), "fr"), on="activite")
)
vraisemblance.filter(pl.col("annee") == ANNEE).sort("energie_tj", descending=True).with_columns(
    pl.col("energie_tj", "achats_mdt").round(0)
)
```

للتكرير (19) والكهرباء (35) سعر ضمني مرتفع، لأن مشترياتهما من المنتجات الطاقوية هي في معظمها مدخلات تحويل، وهي مستبعدة هنا من الاستخدام النهائي. أما الفروع التي تستهلك الطاقة دون أن تشتريها في الجداول، أو العكس، فتستحق الفحص؛ ويدرجها الجدول التالي.

```{python}
#| label: incoherences
vraisemblance.filter(
    ((pl.col("energie_tj") > 0) & (pl.col("achats_mdt") <= 0)) | ((pl.col("energie_tj") <= 0) & (pl.col("achats_mdt") > 0))
).select("annee", "activite", "fr", "energie_tj", "achats_mdt").sort("activite", "annee")
```

::: {.callout-note}
## مسألة مفتوحة
تشير الأسعار الضمنية المرتفعة جدًا أو المنخفضة جدًا إلى توزيع ينبغي مراجعته. تتلقى فروع الخدمات طاقة قطاع الخدمات حسب مشترياتها من الكهرباء والغاز؛ ويعكس سعرها الضمني هذا المفتاح أساسًا. والفروع التي تشتري منتجات نفطية دون أن تتلقى طاقة من الميزان (لمركباتها مثلًا) تتوقف على مفتاح النقل البري. ستُناقش هذه النقاط مع المعهد الوطني للإحصاء خلال البعثة.
:::

## كثافة الطاقة

تقيس كثافة الطاقة في فرع ما نسبة استخدامه النهائي للطاقة (دون مدخلات التحويل) إلى إنتاجه أو قيمته المضافة، بالتيراجول لكل مليون دينار بالأسعار الجارية. تتبع هذه المؤشرات جداول الموارد والاستخدامات بالأسعار الجارية، لذلك يجمع تطورها بين الكميات والأسعار.

```{python}
#| label: intensites
comptes_production = (
    tre.filter(pl.col("bloc") == "valeur_ajoutee", pl.col("ligne").is_in(["P1", "B1G"]))
    .pivot("ligne", index=["annee", "colonne"], values="valeur")
    .rename({"colonne": "activite", "P1": "production_mdt", "B1G": "valeur_ajoutee_mdt"})
)
indicateurs = (
    energie_finale.filter(pl.col("activite") != "MEN")
    .join(comptes_production, on=["annee", "activite"], how="left")
    .with_columns(
        intensite_production=pl.col("energie_tj") / pl.col("production_mdt"),
        intensite_va=pl.col("energie_tj") / pl.col("valeur_ajoutee_mdt"),
    )
    .rename({"energie_tj": "energie_finale_tj"})
    .sort("annee", "activite")
)
(
    indicateurs.filter(pl.col("annee") == ANNEE)
    .join(activites.select(pl.col("code").alias("activite"), "fr"), on="activite")
    .select("activite", "fr", pl.col("energie_finale_tj").round(0), pl.col("intensite_va").round(2))
    .sort("intensite_va", descending=True)
)
```

## التصدير

يحتوي المصنف على ورقة ملاحظات، والجدول الطويل لجميع السنوات، وورقة موارد وورقة استخدامات لكل سنة، والمؤشرات. تأتي تسميات الأسطر من `data/nomenclatures/pefa_lignes.csv`، وتسميات الأنشطة من `data/nomenclatures/activites.csv`، والعناوين من `data/nomenclatures/libelles.csv`. وتتجه الأوراق العربية من اليمين إلى اليسار.

```{python}
#| label: export
ORDRE_ACTIVITES = activites["code"].to_list()
ORDRE_LIGNES = {(t, c): o for t, c, o in lignes.select("type", "cle", "ordre").iter_rows()}


def traduire(langue: str) -> tuple[dict, dict, dict]:
    t = dict(zip(libelles["cle"], libelles[langue]))
    l = dict(zip(lignes["cle"], lignes[langue]))
    a = dict(zip(activites["code"], activites[langue]))
    return t, l, a


def matrice(annee: int, quadrant: str, langue: str) -> pl.DataFrame:
    """Tableau des ressources ou des emplois d'une année : produits en lignes, activités en colonnes."""
    t, l, a = traduire(langue)
    donnees = pefa.filter(pl.col("annee") == annee, pl.col("quadrant") == quadrant)
    presentes = [c for c in ORDRE_ACTIVITES if c in set(donnees["activite"])]
    tableau = (
        donnees.with_columns(
            o1=pl.col("niveau1").replace_strict({c: o for (ty, c), o in ORDRE_LIGNES.items() if ty == "niveau1"}),
            o2=pl.col("niveau2").replace_strict({c: o for (ty, c), o in ORDRE_LIGNES.items() if ty == "niveau2"}, default=0),
            o3=pl.col("produit").replace_strict({c: o for (ty, c), o in ORDRE_LIGNES.items() if ty == "produit"}),
        )
        .pivot("activite", index=["o1", "o2", "o3", "niveau1", "niveau2", "produit"], values="tj", aggregate_function="sum")
        .sort("o1", "o2", "o3")
    )
    tableau = tableau.with_columns(pl.col(presentes).fill_null(0)).with_columns(
        pl.sum_horizontal(presentes).alias("total")
    )
    return tableau.select(
        pl.col("niveau1").replace_strict(l).alias(t["niveau1"]),
        pl.col("niveau2").replace_strict(l, default=None).alias(t["niveau2"]),
        pl.col("produit").replace_strict(l).alias(t["produit"]),
        *[pl.col(c).alias(f"{c} {a[c]}") for c in presentes],
        pl.col("total").alias(t["total"]),
    )


def donnees_longues(langue: str) -> pl.DataFrame:
    t, l, a = traduire(langue)
    return pefa.select(
        pl.col("annee").alias(t["annee"]),
        pl.col("quadrant").replace_strict({"Supply": t["Supply"], "Use": t["Use"]}).alias(t["quadrant"]),
        pl.col("niveau1").replace_strict(l).alias(t["niveau1"]),
        pl.col("niveau2").replace_strict(l, default=None).alias(t["niveau2"]),
        pl.col("produit").replace_strict(l).alias(t["produit"]),
        pl.col("activite").alias(t["code_activite"]),
        pl.col("activite").replace_strict(a).alias(t["activite"]),
        pl.col("tj").alias(t["tj"]),
    )


def indicateurs_traduits(langue: str) -> pl.DataFrame:
    t, l, a = traduire(langue)
    return indicateurs.select(
        pl.col("annee").alias(t["annee"]),
        pl.col("activite").alias(t["code_activite"]),
        pl.col("activite").replace_strict(a).alias(t["activite"]),
        *[pl.col(c).alias(t[c]) for c in ["energie_finale_tj", "production_mdt", "valeur_ajoutee_mdt", "intensite_production", "intensite_va"]],
    )


def exporter(langue: str) -> Path:
    t, _, _ = traduire(langue)
    dossier = PRODUITS / langue
    dossier.mkdir(parents=True, exist_ok=True)
    chemin = dossier / f"{NOM}.xlsx"
    with xlsxwriter.Workbook(chemin) as classeur:
        notes = classeur.add_worksheet(t["feuille_notes"])
        for i, cle in enumerate(["titre_pefa", "unite_tj", "note_source", "note_methode", "note_provisoire", "note_site"]):
            notes.write(i, 0, t[cle])
        feuilles = [(t["feuille_donnees"], donnees_longues(langue), 1)]
        for annee in ANNEES:
            feuilles.append((f"{t['feuille_ressources']} {annee}", matrice(annee, "Supply", langue), 1))
            feuilles.append((f"{t['feuille_emplois']} {annee}", matrice(annee, "Use", langue), 1))
        feuilles.append((t["feuille_indicateurs"], indicateurs_traduits(langue), 3))
        for nom, tableau, decimales in feuilles:
            tableau.write_excel(workbook=classeur, worksheet=nom, autofit=True, float_precision=decimales)
        if langue == "ar":
            for feuille in classeur.worksheets():
                feuille.right_to_left()
    return chemin


for langue in ["fr", "en", "ar"]:
    print(exporter(langue))
```
