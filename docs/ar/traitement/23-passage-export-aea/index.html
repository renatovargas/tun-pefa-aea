2 جدول الربط

تتبع حسابات AEA مبدأ الإقامة، ويتبع الجرد مبدأ الإقليم. ويشرح جدول الربط الفرق بينهما (Eurostat, 2015). فانطلاقًا من حسابات AEA، يطرح انبعاثات المقيمين في الخارج ويضيف انبعاثات غير المقيمين في الإقليم. ويضيف أيضًا التعديلات الأخرى، ثم LULUCF، للوصول إلى مجموع الجرد.

يحتاج النقل البحري للمقيمين في الخارج إلى تصحيح. تضيف الأداة الفارق الموجب بين سفن المقيمين والملاحة المحلية إلى الأنشطة، دون تسجيله في بند الربط (الصفحة 22). ويعيد الجدول بناء هذا البند من الانبعاثات ذات المصدر ALL_VESSELS المسندة إلى الأنشطة. وتتحقق الشيفرة من أن الربط يصل إلى الجرد دون LULUCF في الصفحة 20، غازًا غازًا وسنة سنة.

def par_gaz(table: pl.DataFrame, element: str) -> pl.DataFrame:
    return table.group_by("annee", "gaz").agg(kt=pl.col("kt_eq_co2").sum()).with_columns(element=pl.lit(element))


detail = aea.filter(pl.col("gaz") != "GHG")
elements = pl.concat([
    par_gaz(detail.filter(pl.col("activite") == "AEA-TOTAL"), "AEA"),
    *[par_gaz(detail.filter(pl.col("activite") == e), e) for e in ["NRA-AIR", "NRT-AIR", "NRT-WATER", "OTHER", "LULUCF"]],
    # Écart positif des navires ajouté aux activités par l'outil, sans élément de passage
    par_gaz(detail.filter(pl.col("source") == "ALL_VESSELS", pl.col("activite").is_in(ACTIVITES)), "NRA-WATER"),
    par_gaz(inventaire.filter(pl.col("perimetre") == "inventaire"), "INV"),
])
elements = pl.concat([elements, elements.group_by("annee", "element").agg(pl.col("kt").sum()).with_columns(gaz=pl.lit("GHG")).select(elements.columns)])
passage = (
    elements.pivot("element", index=["annee", "gaz"], values="kt")
    .fill_null(0)
    .with_columns(pl.lit(0.0).alias(c) for c in ["NRA-AIR", "NRA-WATER", "NRT-AIR", "NRT-WATER", "OTHER"] if c not in elements["element"])
    .with_columns(calcul=pl.col("AEA") - pl.col("NRA-AIR") - pl.col("NRA-WATER") + pl.col("NRT-AIR") + pl.col("NRT-WATER") + pl.col("OTHER"))
    .with_columns(ecart=pl.col("calcul") - pl.col("INV"), **{"INV-TOTAL": pl.col("INV") + pl.col("LULUCF")})
    .sort("annee", "gaz")
)
assert passage["ecart"].abs().max() < 1e-6, "Le passage des comptes AEA à l'inventaire ne boucle pas"
print(f"Écart maximal entre le passage et l'inventaire : {passage['ecart'].abs().max():.1e} kt")

ORDRE = passage_lignes.sort("ordre")["code"].to_list()
LIBELLES_PASSAGE = dict(passage_lignes.select("code", "fr").iter_rows())
(
    passage.filter(pl.col("gaz") == "GHG")
    .unpivot(index="annee", on=ORDRE, variable_name="code", value_name="kt")
    .with_columns(pl.col("kt").round(1))
    .pivot("annee", index="code", values="kt")
    .with_columns(element=pl.col("code").replace_strict(LIBELLES_PASSAGE))
    .select("element", pl.exclude("code", "element"))
)

لا يطابق المجموع INV-TOTAL في الأداة الجرد. فالأداة تجمع بنود الربط دون مراعاة إشارتها. ومع أن جدول أنشطتها يحمل الإشارة «-» للمقيمين في الخارج، وأن الشيفرة تقرؤها (الأسطر 521 إلى 524 من I2A.py)، فإن حاصل الضرب لا يُستخدم في المجاميع. لذلك يتجاوز مجموع الأداة الجرد بضعف النقل الجوي للمقيمين في الخارج، مضافًا إليه النقل البحري الذي لم تسجله الأداة بندًا للربط. ويتحقق الجدول التالي من ذلك.

total_outil = aea.filter(pl.col("gaz") == "GHG", pl.col("activite") == "INV-TOTAL").select("annee", outil="kt_eq_co2")
(
    passage.filter(pl.col("gaz") == "GHG")
    .join(total_outil, on="annee")
    .select(
        "annee",
        pl.col("INV-TOTAL").alias("inventaire"),
        "outil",
        (pl.col("outil") - pl.col("INV-TOTAL")).alias("ecart"),
        (2 * pl.col("NRA-AIR") + pl.col("NRA-WATER")).alias("deux_nra_air_plus_nra_water"),
    )
    .with_columns(pl.exclude("annee").round(1))
)

3 المقيمون في الخارج والمستودعات الدولية

تعطي بيانات منظمة التعاون والتنمية في الميدان الاقتصادي انبعاثات الناقلين المقيمين، أينما تزودوا بالوقود. وتعطي المستودعات الدولية في الجرد (البنود التذكيرية 1.D) انبعاثات الوقود المبيع في تونس للناقلين الدوليين، مقيمين كانوا أم غير مقيمين. يختلف المفهومان، لكن يمكن مقارنة رتبتي حجمهما. في النقل الجوي، هما من الرتبة نفسها. وفي النقل البحري، تُصدر سفن المقيمين أكثر بكثير من المستودعات البحرية المبيعة في تونس. ولا تبدأ البيانات البحرية للمنظمة إلا في 2019.

4 المعقولية بالمقارنة مع حسابات PEFA

تأتي انبعاثات CO2 الناتجة عن الاحتراق (الفئة 1.A) في فرع ما من أنواع الوقود الأحفوري التي يستخدمها. وتعطي حسابات PEFA هذا الاستخدام بالتيراجول (الصفحة 14). والنسبة بين الاثنين معامل انبعاث ضمني، بأطنان CO2 لكل تيراجول. ويجب أن تبقى ضمن نطاق المعاملات الضمنية للجرد نفسه، المحسوبة حسب نوع الوقود على الاحتراق في الفئة 1.A، ببيانات النشاط بالتيراجول الواردة في جداول CRT (Intergovernmental Panel on Climate Change, 2006).

crt = pl.read_parquet("data/intermediaires/crt.parquet").filter(pl.col("annee") == 2022, pl.col("crt").str.starts_with("1.A."))
activite_tj = crt.filter(pl.col("parametre") == "activity", pl.col("unite") == "TJ").group_by("combustible").agg(tj=pl.col("valeur").sum())
co2_kt = crt.filter(pl.col("parametre") == "emission", pl.col("gaz") == "CO2").group_by("combustible").agg(co2_kt=pl.col("valeur").sum())
(
    activite_tj.join(co2_kt, on="combustible")
    .filter(pl.col("tj") > 0, ~pl.col("combustible").is_in(["biomass"]))
    .with_columns(t_co2_par_tj=(1000 * pl.col("co2_kt") / pl.col("tj")).round(1), tj=pl.col("tj").round(0))
    .sort("t_co2_par_tj")
)

يدل الفارق على توزيع مختلف في الحسابين. فالاستخدام غير الطاقوي للمنتجات النفطية، المحتسب في حسابات PEFA دون انبعاثات احتراق، يخفض النسبة. وترفعها تعديلات الإقامة للنقل الجوي والبحري، الغائبة عن حسابات PEFA، في فرع النقل.

COMBUSTIBLES_FOSSILES = ["Coal", "Peat and peat products", "Oil shale / oil sands", "Natural gas", "Oil"]
energie = (
    pefa.filter(
        pl.col("quadrant") == "Use",
        pl.col("niveau2") == "End-use of energy products by SIEC class:",
        pl.col("produit").is_in(COMBUSTIBLES_FOSSILES),
        pl.col("activite").is_in(ACTIVITES),
    )
    .group_by("annee", "activite")
    .agg(energie_tj=pl.col("tj").sum())
)
co2_combustion = (
    aea.filter(pl.col("gaz") == "CO2", pl.col("source").str.starts_with("1.A."), pl.col("activite").is_in(ACTIVITES))
    .group_by("annee", "activite")
    .agg(co2_kt=pl.col("kt_eq_co2").sum())
)
facteurs = (
    co2_combustion.join(energie, on=["annee", "activite"], how="full", coalesce=True)
    .with_columns(pl.col("co2_kt", "energie_tj").fill_null(0))
    .with_columns(t_co2_par_tj=pl.when(pl.col("energie_tj") > 0).then(1000 * pl.col("co2_kt") / pl.col("energie_tj")))
    .rename({"energie_tj": "combustibles_fossiles_tj"})
)
(
    facteurs.filter(pl.col("annee") == 2022)
    .with_columns(libelle=pl.col("activite").replace_strict(LIBELLES_FR, default=None))
    .select("activite", "libelle", pl.col("co2_kt").round(1), pl.col("combustibles_fossiles_tj").round(0), pl.col("t_co2_par_tj").round(1))

5 كثافة الانبعاثات

تنسب كثافة الانبعاثات لفرع ما انبعاثاته من غازات الاحتباس الحراري إلى إنتاجه أو إلى قيمته المضافة، بأطنان مكافئ ثاني أكسيد الكربون لكل مليون دينار بالأسعار الجارية. وفي الأسر، المقام هو نفقات الاستهلاك النهائي، كما يقترح دليل الأداة (International Monetary Fund, 2026). وتتبع هذه المؤشرات جداول الموارد والاستخدامات بالأسعار الجارية، فيمزج تطورها بين الأحجام والأسعار.

comptes_production = (
    tre.filter(pl.col("bloc") == "valeur_ajoutee", pl.col("ligne").is_in(["P1", "B1G"]))
    .pivot("ligne", index=["annee", "colonne"], values="valeur")
    .rename({"colonne": "activite", "P1": "production_mdt", "B1G": "valeur_ajoutee_mdt"})
)
depense_menages = (
    tre.filter(pl.col("bloc") == "emplois", pl.col("colonne") == "P3_MEN", ~pl.col("agregat"))
    .group_by("annee")
    .agg(depense_men_mdt=pl.col("valeur").sum())
    .with_columns(activite=pl.lit("MEN"))
)
ges = (
    aea.filter(pl.col("gaz") == "GHG", pl.col("activite").is_in(ACTIVITES))
    .group_by("annee", "activite")
    .agg(emissions_ges_kt=pl.col("kt_eq_co2").sum())
)
indicateurs = (
    ges.join(comptes_production, on=["annee", "activite"], how="left")
    .join(depense_menages, on=["annee", "activite"], how="left")
    .with_columns(
        intensite_ges_production=1000 * pl.col("emissions_ges_kt") / pl.col("production_mdt"),
        intensite_ges_va=1000 * pl.col("emissions_ges_kt") / pl.col("valeur_ajoutee_mdt"),
        intensite_ges_men=1000 * pl.col("emissions_ges_kt") / pl.col("depense_men_mdt"),
    )
    .sort("annee", "activite")
)
(
    indicateurs.filter(pl.col("annee") == 2022)
    .with_columns(libelle=pl.col("activite").replace_strict(LIBELLES_FR))
    .select("activite", "libelle", pl.col("emissions_ges_kt").round(1), pl.col("intensite_ges_va", "intensite_ges_men").round(1))
    .sort("intensite_ges_va", descending=True, nulls_last=True)
)

6 التصدير

يحتوي المصنف على ورقة ملاحظات، والجدول الطويل لجميع السنوات مع فئة CRT الأصلية، وجدول الربط، وورقة لكل سنة فيها الأنشطة في الأسطر والغازات في الأعمدة، والمؤشرات. تأتي تسميات الأنشطة من data/nomenclatures/activites.csv، وتسميات الغازات من data/nomenclatures/gaz.csv، وتسميات الربط من data/nomenclatures/aea_passage.csv، والعناوين من data/nomenclatures/libelles.csv. وتُعرض الأوراق العربية من اليمين إلى اليسار.

def traduire(langue: str) -> tuple[dict, dict, dict, dict]:
    t = dict(zip(libelles["cle"], libelles[langue]))
    a = dict(zip(activites["code"], activites[langue]))
    g = dict(zip(gaz["code"], gaz[langue]))
    p = dict(zip(passage_lignes["code"], passage_lignes[langue]))
    return t, a, g, p


def tableau_annee(annee: int, langue: str) -> pl.DataFrame:
    """Émissions d'une année : activités en lignes, gaz en colonnes, avec le total."""
    t, a, g, _ = traduire(langue)
    donnees = aea.filter(pl.col("annee") == annee, pl.col("activite").is_in(ACTIVITES))
    large = (
        donnees.group_by("activite", "gaz").agg(pl.col("kt_eq_co2").sum())
        .pivot("gaz", index="activite", values="kt_eq_co2")
        .with_columns(ordre=pl.col("activite").replace_strict({c: i for i, c in enumerate(ACTIVITES)}, return_dtype=pl.Int16))
        .sort("ordre")
    )
    colonnes = [c for c in GAZ + ["GHG"] if c in large.columns]
    large = large.with_columns(pl.col(colonnes).fill_null(0))
    total = large.select(pl.col(colonnes).sum()).with_columns(activite=pl.lit(None, dtype=pl.String), ordre=pl.lit(999, dtype=pl.Int16))
    large = pl.concat([large, total.select(large.columns)])
    return large.select(
        pl.col("activite").alias(t["code_activite"]),
        pl.col("activite").replace_strict(a, default=t["total"]).alias(t["activite"]),
        *[pl.col(c).alias(g[c]) for c in colonnes],
    )


def passage_traduit(langue: str) -> pl.DataFrame:
    t, _, _, p = traduire(langue)
    return (
        passage.filter(pl.col("gaz") == "GHG")
        .unpivot(index="annee", on=ORDRE, variable_name="code", value_name="kt")
        .pivot("annee", index="code", values="kt")
        .select(pl.col("code").replace_strict(p).alias(t["element_passage"]), pl.exclude("code"))
    )


def donnees_longues(langue: str) -> pl.DataFrame:
    t, a, g, _ = traduire(langue)
    return aea.filter(pl.col("activite").is_in(ACTIVITES)).select(
        pl.col("annee").alias(t["annee"]),
        pl.col("gaz").replace_strict(g).alias(t["gaz"]),
        pl.col("source").alias(t["categorie_crt"]),
        pl.col("activite").alias(t["code_activite"]),
        pl.col("activite").replace_strict(a).alias(t["activite"]),
        pl.col("kt_eq_co2").alias(t["kt_eq_co2"]),
    )


def indicateurs_traduits(langue: str) -> pl.DataFrame:
    t, a, _, _ = traduire(langue)
    colonnes = ["emissions_ges_kt", "production_mdt", "valeur_ajoutee_mdt", "depense_men_mdt",
                "intensite_ges_production", "intensite_ges_va", "intensite_ges_men"]
    return indicateurs.select(
        pl.col("annee").alias(t["annee"]),
        pl.col("activite").alias(t["code_activite"]),
        pl.col("activite").replace_strict(a).alias(t["activite"]),
        *[pl.col(c).alias(t[c]) for c in colonnes],
    )


def exporter(langue: str) -> Path:
    t, _, _, _ = traduire(langue)
    dossier = PRODUITS / langue
    dossier.mkdir(parents=True, exist_ok=True)
    chemin = dossier / f"{NOM}.xlsx"
    with xlsxwriter.Workbook(chemin) as classeur:
        notes = classeur.add_worksheet(t["feuille_notes"])
        for i, cle in enumerate(["titre_aea", "unite_kt_eq", "note_source_aea", "note_methode_aea", "note_provisoire", "note_site"]):
            notes.write(i, 0, t[cle])
        feuilles = [(t["feuille_donnees"], donnees_longues(langue), 3), (t["feuille_passage"], passage_traduit(langue), 1)]
        feuilles += [(f"{t['feuille_aea']} {annee}", tableau_annee(annee, langue), 1) for annee in ANNEES]
        feuilles.append((t["feuille_indicateurs"], indicateurs_traduits(langue), 1))
        for nom, tableau, decimales in feuilles:
            tableau.write_excel(workbook=classeur, worksheet=nom, autofit=True, float_precision=decimales)
        if langue == "ar":
            for feuille in classeur.worksheets():
                feuille.right_to_left()
    return chemin

## جدول الربط

تتبع حسابات AEA مبدأ الإقامة، ويتبع الجرد مبدأ الإقليم. ويشرح جدول الربط الفرق بينهما [@eurostat2015aea]. فانطلاقًا من حسابات AEA، يطرح انبعاثات المقيمين في الخارج ويضيف انبعاثات غير المقيمين في الإقليم. ويضيف أيضًا التعديلات الأخرى، ثم LULUCF، للوصول إلى مجموع الجرد.

يحتاج النقل البحري للمقيمين في الخارج إلى تصحيح. تضيف الأداة الفارق الموجب بين سفن المقيمين والملاحة المحلية إلى الأنشطة، دون تسجيله في بند الربط (الصفحة 22). ويعيد الجدول بناء هذا البند من الانبعاثات ذات المصدر `ALL_VESSELS` المسندة إلى الأنشطة. وتتحقق الشيفرة من أن الربط يصل إلى الجرد دون LULUCF في الصفحة 20، غازًا غازًا وسنة سنة.

```{python}
#| label: passage
def par_gaz(table: pl.DataFrame, element: str) -> pl.DataFrame:
    return table.group_by("annee", "gaz").agg(kt=pl.col("kt_eq_co2").sum()).with_columns(element=pl.lit(element))


detail = aea.filter(pl.col("gaz") != "GHG")
elements = pl.concat([
    par_gaz(detail.filter(pl.col("activite") == "AEA-TOTAL"), "AEA"),
    *[par_gaz(detail.filter(pl.col("activite") == e), e) for e in ["NRA-AIR", "NRT-AIR", "NRT-WATER", "OTHER", "LULUCF"]],
    # Écart positif des navires ajouté aux activités par l'outil, sans élément de passage
    par_gaz(detail.filter(pl.col("source") == "ALL_VESSELS", pl.col("activite").is_in(ACTIVITES)), "NRA-WATER"),
    par_gaz(inventaire.filter(pl.col("perimetre") == "inventaire"), "INV"),
])
elements = pl.concat([elements, elements.group_by("annee", "element").agg(pl.col("kt").sum()).with_columns(gaz=pl.lit("GHG")).select(elements.columns)])
passage = (
    elements.pivot("element", index=["annee", "gaz"], values="kt")
    .fill_null(0)
    .with_columns(pl.lit(0.0).alias(c) for c in ["NRA-AIR", "NRA-WATER", "NRT-AIR", "NRT-WATER", "OTHER"] if c not in elements["element"])
    .with_columns(calcul=pl.col("AEA") - pl.col("NRA-AIR") - pl.col("NRA-WATER") + pl.col("NRT-AIR") + pl.col("NRT-WATER") + pl.col("OTHER"))
    .with_columns(ecart=pl.col("calcul") - pl.col("INV"), **{"INV-TOTAL": pl.col("INV") + pl.col("LULUCF")})
    .sort("annee", "gaz")
)
assert passage["ecart"].abs().max() < 1e-6, "Le passage des comptes AEA à l'inventaire ne boucle pas"
print(f"Écart maximal entre le passage et l'inventaire : {passage['ecart'].abs().max():.1e} kt")

ORDRE = passage_lignes.sort("ordre")["code"].to_list()
LIBELLES_PASSAGE = dict(passage_lignes.select("code", "fr").iter_rows())
(
    passage.filter(pl.col("gaz") == "GHG")
    .unpivot(index="annee", on=ORDRE, variable_name="code", value_name="kt")
    .with_columns(pl.col("kt").round(1))
    .pivot("annee", index="code", values="kt")
    .with_columns(element=pl.col("code").replace_strict(LIBELLES_PASSAGE))
    .select("element", pl.exclude("code", "element"))
)
```

## المعقولية بالمقارنة مع حسابات PEFA

تأتي انبعاثات CO2 الناتجة عن الاحتراق (الفئة 1.A) في فرع ما من أنواع الوقود الأحفوري التي يستخدمها. وتعطي حسابات PEFA هذا الاستخدام بالتيراجول (الصفحة 14). والنسبة بين الاثنين معامل انبعاث ضمني، بأطنان CO2 لكل تيراجول. ويجب أن تبقى ضمن نطاق المعاملات الضمنية للجرد نفسه، المحسوبة حسب نوع الوقود على الاحتراق في الفئة 1.A، ببيانات النشاط بالتيراجول الواردة في جداول CRT [@ipcc2006].

## كثافة الانبعاثات

تنسب كثافة الانبعاثات لفرع ما انبعاثاته من غازات الاحتباس الحراري إلى إنتاجه أو إلى قيمته المضافة، بأطنان مكافئ ثاني أكسيد الكربون لكل مليون دينار بالأسعار الجارية. وفي الأسر، المقام هو نفقات الاستهلاك النهائي، كما يقترح دليل الأداة [@imf2026i2a]. وتتبع هذه المؤشرات جداول الموارد والاستخدامات بالأسعار الجارية، فيمزج تطورها بين الأحجام والأسعار.

## التصدير

يحتوي المصنف على ورقة ملاحظات، والجدول الطويل لجميع السنوات مع فئة CRT الأصلية، وجدول الربط، وورقة لكل سنة فيها الأنشطة في الأسطر والغازات في الأعمدة، والمؤشرات. تأتي تسميات الأنشطة من `data/nomenclatures/activites.csv`، وتسميات الغازات من `data/nomenclatures/gaz.csv`، وتسميات الربط من `data/nomenclatures/aea_passage.csv`، والعناوين من `data/nomenclatures/libelles.csv`. وتُعرض الأوراق العربية من اليمين إلى اليسار.

```{python}
#| label: export
def traduire(langue: str) -> tuple[dict, dict, dict, dict]:
    t = dict(zip(libelles["cle"], libelles[langue]))
    a = dict(zip(activites["code"], activites[langue]))
    g = dict(zip(gaz["code"], gaz[langue]))
    p = dict(zip(passage_lignes["code"], passage_lignes[langue]))
    return t, a, g, p


def tableau_annee(annee: int, langue: str) -> pl.DataFrame:
    """Émissions d'une année : activités en lignes, gaz en colonnes, avec le total."""
    t, a, g, _ = traduire(langue)
    donnees = aea.filter(pl.col("annee") == annee, pl.col("activite").is_in(ACTIVITES))
    large = (
        donnees.group_by("activite", "gaz").agg(pl.col("kt_eq_co2").sum())
        .pivot("gaz", index="activite", values="kt_eq_co2")
        .with_columns(ordre=pl.col("activite").replace_strict({c: i for i, c in enumerate(ACTIVITES)}, return_dtype=pl.Int16))
        .sort("ordre")
    )
    colonnes = [c for c in GAZ + ["GHG"] if c in large.columns]
    large = large.with_columns(pl.col(colonnes).fill_null(0))
    total = large.select(pl.col(colonnes).sum()).with_columns(activite=pl.lit(None, dtype=pl.String), ordre=pl.lit(999, dtype=pl.Int16))
    large = pl.concat([large, total.select(large.columns)])
    return large.select(
        pl.col("activite").alias(t["code_activite"]),
        pl.col("activite").replace_strict(a, default=t["total"]).alias(t["activite"]),
        *[pl.col(c).alias(g[c]) for c in colonnes],
    )


def passage_traduit(langue: str) -> pl.DataFrame:
    t, _, _, p = traduire(langue)
    return (
        passage.filter(pl.col("gaz") == "GHG")
        .unpivot(index="annee", on=ORDRE, variable_name="code", value_name="kt")
        .pivot("annee", index="code", values="kt")
        .select(pl.col("code").replace_strict(p).alias(t["element_passage"]), pl.exclude("code"))
    )


def donnees_longues(langue: str) -> pl.DataFrame:
    t, a, g, _ = traduire(langue)
    return aea.filter(pl.col("activite").is_in(ACTIVITES)).select(
        pl.col("annee").alias(t["annee"]),
        pl.col("gaz").replace_strict(g).alias(t["gaz"]),
        pl.col("source").alias(t["categorie_crt"]),
        pl.col("activite").alias(t["code_activite"]),
        pl.col("activite").replace_strict(a).alias(t["activite"]),
        pl.col("kt_eq_co2").alias(t["kt_eq_co2"]),
    )


def indicateurs_traduits(langue: str) -> pl.DataFrame:
    t, a, _, _ = traduire(langue)
    colonnes = ["emissions_ges_kt", "production_mdt", "valeur_ajoutee_mdt", "depense_men_mdt",
                "intensite_ges_production", "intensite_ges_va", "intensite_ges_men"]
    return indicateurs.select(
        pl.col("annee").alias(t["annee"]),
        pl.col("activite").alias(t["code_activite"]),
        pl.col("activite").replace_strict(a).alias(t["activite"]),
        *[pl.col(c).alias(t[c]) for c in colonnes],
    )


def exporter(langue: str) -> Path:
    t, _, _, _ = traduire(langue)
    dossier = PRODUITS / langue
    dossier.mkdir(parents=True, exist_ok=True)
    chemin = dossier / f"{NOM}.xlsx"
    with xlsxwriter.Workbook(chemin) as classeur:
        notes = classeur.add_worksheet(t["feuille_notes"])
        for i, cle in enumerate(["titre_aea", "unite_kt_eq", "note_source_aea", "note_methode_aea", "note_provisoire", "note_site"]):
            notes.write(i, 0, t[cle])
        feuilles = [(t["feuille_donnees"], donnees_longues(langue), 3), (t["feuille_passage"], passage_traduit(langue), 1)]
        feuilles += [(f"{t['feuille_aea']} {annee}", tableau_annee(annee, langue), 1) for annee in ANNEES]
        feuilles.append((t["feuille_indicateurs"], indicateurs_traduits(langue), 1))
        for nom, tableau, decimales in feuilles:
            tableau.write_excel(workbook=classeur, worksheet=nom, autofit=True, float_precision=decimales)
        if langue == "ar":
            for feuille in classeur.worksheets():
                feuille.right_to_left()
    return chemin


for langue in ["fr", "en", "ar"]:
    print(exporter(langue))
```
