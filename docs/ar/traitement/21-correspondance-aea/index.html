1 الغرض

يصنّف الجرد الانبعاثات حسب العملية، كالاحتراق في صناعات الطاقة والنقل البري والتخمر المعوي. أما حسابات AEA فتصنّفها حسب الوحدة الاقتصادية التي تنتجها، أي فروع النشاط في الحسابات القومية والأسر (Eurostat, 2015). وتقوم أداة I2A لصندوق النقد الدولي (International Monetary Fund, 2026; Pegoue, 2026) بهذا الانتقال بواسطة مصفوفة تحويل. تعطي المصفوفة، لكل فئة CRT ولكل غاز، حصة الانبعاثات المسندة إلى كل نشاط.

تفحص هذه الصفحة المصفوفة الافتراضية للأداة، ثم تبني مصفوفة تونسية حسب فروع جداول الموارد والاستخدامات، على غرار مصفوفة التوافق لحسابات PEFA (الصفحة 12). وبذلك يتقاسم الحسابان الفروع نفسها ومفاتيح التوزيع نفسها.

import sys

import polars as pl

sys.path.insert(0, "scripts")
import outils_fmi as of

SORTIE = "data/intermediaires/matrice_aea_tunisie.parquet"
SORTIE_DIFFERENCES = "data/intermediaires/matrice_aea_differences.parquet"
CODES = {"code": pl.String, "activite": pl.String, "prefixe": pl.String}

inventaire = pl.read_parquet("data/intermediaires/inventaire_ges.parquet").filter(pl.col("perimetre") != "pour_memoire")
cles_pefa = pl.read_parquet("data/intermediaires/cles_activites.parquet")
activites = pl.read_csv("data/nomenclatures/activites.csv", schema_overrides=CODES)
correspondance = pl.read_csv("data/nomenclatures/crt_vers_activites.csv", schema_overrides=CODES)
plages = of.plages_i2a()
ANNEES = sorted(inventaire["annee"].unique().to_list())

2 المصفوفة الافتراضية للأداة

يحتوي مصنف الأداة على مجموعتين من الجداول. تحمل الأوراق النشطة معاملات توضيحية، مسحوبة عشوائيًا، لمثال كندا. أما الجداول الافتراضية فهي في الأوراق التي ينتهي اسمها بـ«_Def». تقرأ الوحدة scripts/outils_fmi.py1 هذه الأخيرة، بتعريفات النطاقات نفسها التي تستخدمها النطاقات المسماة في المصنف.

للمصفوفة الافتراضية سطر لكل غاز ولكل فئة CRT، وعمود لكل نشاط. والأنشطة هي أقسام التصنيف الأوروبي NACE، القريب من التصنيف الصناعي الدولي الموحد (ISIC)، والأسر حسب غرض الاستهلاك، وبنود الربط.

entete = [str(c) for c in plages["Conversion_Matrix"][0]]
ACTIVITES_OUTIL = entete[5:]
matrice_defaut = (
    pl.DataFrame(plages["Conversion_Matrix"][1:], schema=entete, orient="row", infer_schema_length=None)
    .unpivot(index=["POLLUTANT", "Source_Code", "Check"], on=ACTIVITES_OUTIL, variable_name="activite_outil", value_name="facteur")
    .with_columns(pl.col("facteur").cast(pl.Float64).fill_null(0), pl.col("Check").cast(pl.Float64).fill_null(0))
)


def section_outil(activite: str) -> str:
    """Section de la CITI d'une activité de l'outil, dans les codes de section du site."""
    if activite.startswith("HH"):
        return "HH"
    if activite[0] in "ABCDEFH" and not activite.startswith(("NRA", "NRT")):
        return activite[0]
    if activite in ("LULUCF", "OTHER") or activite.startswith(("NRA", "NRT")):
        return activite
    return "Other"


print(f"{matrice_defaut['Source_Code'].n_unique()} catégories, {len(ACTIVITES_OUTIL)} activités, gaz : {sorted(matrice_defaut['POLLUTANT'].unique())}")
matrice_defaut.filter(pl.col("POLLUTANT") == "CO2", pl.col("facteur") > 0, pl.col("Source_Code").is_in(["1.A.3.b", "1.A.4.a", "1.A.4.b", "1.A.4.c", "3.A", "5.A"]))

لا تحمل المصفوفة الافتراضية أي معامل لبعض الفئات. فتترك الأداة انبعاثاتها دون نشاط، وتختفي من الحسابات دون أي رسالة. وهذه حال التربة الزراعية (3.D)، التي تُعدّ انبعاثاتها من N2O كبيرة في تونس.

sans_facteur = matrice_defaut.filter(pl.col("Check") == 0).select(gaz="POLLUTANT", code_outil="Source_Code").unique()
(
    inventaire.filter(pl.col("perimetre") == "inventaire")
    .join(sans_facteur, on=["gaz", "code_outil"])
    .group_by("annee", "code_outil")
    .agg(pl.col("kt_eq_co2").sum().round(1))
    .sort("annee")
    .pivot("annee", index="code_outil", values="kt_eq_co2")
    .sort("code_outil")
)

3 المقابلة التونسية

يربط الجدول data/nomenclatures/crt_vers_activites.csv بادئات رموز CRT بالأنشطة الواردة في data/nomenclatures/activites.csv، أي فروع جداول الموارد والاستخدامات الأربعة والعشرين والأسر. تأخذ كل فئة من فئات الجرد السطر ذا البادئة الأطول. فتغطي البادئة القصيرة («3» للفلاحة) قطاعًا كاملًا، وتصحح البادئة الطويلة («1.B.2.a.iv» لتكرير النفط وتخزينه) حالة خاصة. ويشرح العمود justification كل اختيار. وتذهب انبعاثات LULUCF إلى النشاط الشكلي LULUCF في الأداة، خارج الحسابات.

correspondance.select("prefixe", "activite", "methode", "cle", "justification")

يجب أن تجد كل فئة من فئات الجرد سطرًا. وتحتاج رموز الطيران المحلي والملاحة المحلية إلى سطر أيضًا، حتى دون انبعاثات في الجرد، لأن الأداة تُدخل عبرها بيانات منظمة التعاون والتنمية في الميدان الاقتصادي عن نقل المقيمين (الصفحة 22). ودون سطر في المصفوفة، تضيع هذه البيانات.

PREFIXES = correspondance["prefixe"].to_list()


def regle(code: str) -> str | None:
    """Préfixe le plus long de la correspondance qui couvre un code (« 3. » est couvert par « 3 »)."""
    code = code.rstrip(".")
    couvrants = [p for p in PREFIXES if code == p or code.startswith(p + ".")]
    return max(couvrants, key=len) if couvrants else None


# Les données de l'OCDE entrent par les codes de l'aviation et de la navigation intérieures
CODES_OCDE = [l[1] for l in plages["CRT"][1:] if l[11] in ("Domestic Aviation", "Domestic Navigation")]
codes = pl.concat([inventaire.select("code_outil"), pl.DataFrame({"code_outil": CODES_OCDE})]).unique().with_columns(
    prefixe=pl.col("code_outil").map_elements(regle, return_dtype=pl.String)
)
assert codes["prefixe"].null_count() == 0, f"Codes sans correspondance : {codes.filter(pl.col('prefixe').is_null())['code_outil'].to_list()}"
print(f"{codes.height} codes de l'outil (inventaire et OCDE : {CODES_OCDE}), tous couverts par {codes['prefixe'].n_unique()} lignes de la correspondance")

4 مفاتيح التوزيع

تجمع فئتان عدة فروع. فالنقل البري (1.A.3.b) تقوم به جميع الفروع والأسر. ويغطي الاحتراق في التجارة والإدارات (1.A.4.a) جميع فروع الخدمات. توزّع حسابات PEFA التدفقات المقابلة من الميزان بمفاتيح مستمدة من جداول الموارد والاستخدامات (الصفحة 11). وتعيد حسابات AEA استخدام هذه المفاتيح، لكي تتبع الانبعاثات الطاقة. وتتبع أجهزة تكييف الهواء في المركبات وأجهزة التكييف الثابتة (2.F.1.e و2.F.1.f) المفاتيح نفسها.

cles = (
    cles_pefa.filter(pl.col("methode") == "cle")
    .select("annee", cle="flux_modele", activite="activite", facteur="facteur")
    .group_by("annee", "cle", "activite")
    .agg(pl.col("facteur").sum())
)
assert set(correspondance["cle"].drop_nulls()) <= set(cles["cle"]), "Clé inconnue dans la correspondance"
cles.filter(pl.col("annee") == 2022).with_columns((pl.col("facteur") * 100).round(1)).pivot("cle", index="activite", values="facteur").sort("activite")

5 المصفوفة التونسية

للمصفوفة التونسية سطر لكل سنة ورمز من رموز الأداة ونشاط. يساوي المعامل 1 في الإسناد المباشر، ويساوي مفتاح السنة في التوزيع. ومجموع معاملات الرمز الواحد يساوي 1. وتصلح المصفوفة لجميع الغازات، لأن القواعد لا تتوقف على الغاز.

annees = pl.DataFrame({"annee": ANNEES}, schema={"annee": pl.Int16})
regles = codes.join(correspondance, on="prefixe")
directes = regles.filter(pl.col("methode") == "direct").join(annees, how="cross").with_columns(facteur=pl.lit(1.0))
reparties = regles.filter(pl.col("methode") == "cle").drop("activite").join(cles, on="cle")
COLONNES = ["annee", "code_outil", "activite", "facteur", "prefixe", "methode", "cle"]
matrice = pl.concat([directes.select(COLONNES), reparties.select(COLONNES)]).sort("annee", "code_outil", "activite")

sommes = matrice.group_by("annee", "code_outil").agg(pl.col("facteur").sum())
assert (sommes["facteur"] - 1).abs().max() < 1e-9, "Les facteurs d'un code doivent sommer à 1"
assert set(matrice["activite"]) <= set(activites["code"]) | {"LULUCF"}, "Activité inconnue"
print(f"{matrice.height} lignes, {matrice['code_outil'].n_unique()} codes, {matrice['activite'].n_unique()} activités")
matrice.filter(pl.col("annee") == 2022, pl.col("code_outil").str.starts_with("1.A.3"))

6 الاختلافات عن المصفوفة الافتراضية

يقارن الجدول التالي، للفئات التي تُصدر انبعاثات في 2022، القسم الرئيسي من التصنيف الصناعي الدولي الموحد في المصفوفة الافتراضية بنظيره في المصفوفة التونسية. تغيّر الأسطر الموسومة بـoui (نعم) في العمود section_modifiee النتيجة حتى على مستوى الأقسام. أما الأسطر الأخرى فلا تضيف سوى التفصيل حسب الفرع.

SECTION_SITE = dict(activites.select("code", "section_fmi").iter_rows()) | {"LULUCF": "LULUCF"}
emissions_2022 = (
    inventaire.filter(pl.col("annee") == 2022)
    .group_by("code_outil")
    .agg(pl.col("kt_eq_co2").sum())
    .filter(pl.col("kt_eq_co2").abs() > 0.05)
)
defaut = (
    matrice_defaut.filter(pl.col("facteur") > 0)
    .join(inventaire.filter(pl.col("annee") == 2022).select(POLLUTANT="gaz", Source_Code="code_outil").unique(), on=["POLLUTANT", "Source_Code"])
    .with_columns(section=pl.col("activite_outil").map_elements(section_outil, return_dtype=pl.String))
    .group_by("Source_Code", "section")
    .agg(pl.col("facteur").sum())
    .sort("facteur", descending=True)
    .group_by("Source_Code", maintain_order=True)
    .agg(section_defaut=pl.col("section").first(), activites_defaut=pl.col("section").unique().sort().str.join(", "))
    .rename({"Source_Code": "code_outil"})
)
tunisie = (
    matrice.filter(pl.col("annee") == 2022)
    .with_columns(section=pl.col("activite").replace_strict(SECTION_SITE))
    .group_by("code_outil", "prefixe")
    .agg(
        activites_tunisie=pl.when(pl.col("methode").first() == "cle")
        .then(pl.lit("clé ") + pl.col("cle").first())
        .otherwise(pl.col("activite").unique().sort().str.join(", ")),
        section_tunisie=pl.col("section").sort_by("facteur", descending=True).first(),
    )
)
differences = (
    emissions_2022.join(tunisie, on="code_outil", how="left")
    .join(defaut, on="code_outil", how="left")
    .with_columns(
        section_modifiee=pl.when(pl.col("section_defaut") != pl.col("section_tunisie")).then(pl.lit("oui"))
        .when(pl.col("section_defaut").is_null()).then(pl.lit("sans facteur"))
        .otherwise(pl.lit("non")),
        kt_eq_co2_2022=pl.col("kt_eq_co2").round(1),
    )
    .sort("code_outil")
    .select("code_outil", "kt_eq_co2_2022", "section_defaut", "activites_defaut", "section_tunisie", "activites_tunisie", "section_modifiee")
)
with pl.Config(tbl_rows=-1, fmt_str_lengths=60):
    print(differences.filter(pl.col("section_modifiee") != "non"))
print(differences.group_by("section_modifiee").agg(pl.len(), pl.col("kt_eq_co2_2022").sum().round(1)).sort("section_modifiee"))
ملاحظةمسألة مفتوحة

يتعين التحقق من الاختيارات التي تبتعد عن المصفوفة الافتراضية مع المعهد الوطني للإحصاء والوكالة الوطنية للتحكم في الطاقة. وأهمها الاحتراق في التجارة والإدارات، المسند إلى الخدمات لا إلى البناء؛ والاحتراق السكني، المسند إلى الأسر؛ والنقل البري، الموزع بين الفروع والأسر بدلًا من إسناده إلى النقل؛ والانبعاثات المتسربة من نقل الغاز وتوزيعه، المسندة إلى فرع الكهرباء والغاز. وينفرد الجرد التونسي بنقطتين. تحمل الفئة 1.A.1.c.i (صنع الوقود الصلب) انبعاثات كبيرة، وتُسند إلى استخراج المحروقات مع بقية الفئة 1.A.1.c. وتحمل الفئة 1.A.4.b.ii (الآلات المتنقلة في القطاع السكني) كل احتراق الأسر، في كل سنة.

للمصفوفة الافتراضية سطر لكل غاز ولكل فئة CRT، وعمود لكل نشاط. والأنشطة هي أقسام التصنيف الأوروبي NACE، القريب من التصنيف الصناعي الدولي الموحد (ISIC)، والأسر حسب غرض الاستهلاك، وبنود الربط.

```{python}
#| label: matrice-defaut
entete = [str(c) for c in plages["Conversion_Matrix"][0]]
ACTIVITES_OUTIL = entete[5:]
matrice_defaut = (
    pl.DataFrame(plages["Conversion_Matrix"][1:], schema=entete, orient="row", infer_schema_length=None)
    .unpivot(index=["POLLUTANT", "Source_Code", "Check"], on=ACTIVITES_OUTIL, variable_name="activite_outil", value_name="facteur")
    .with_columns(pl.col("facteur").cast(pl.Float64).fill_null(0), pl.col("Check").cast(pl.Float64).fill_null(0))
)


def section_outil(activite: str) -> str:
    """Section de la CITI d'une activité de l'outil, dans les codes de section du site."""
    if activite.startswith("HH"):
        return "HH"
    if activite[0] in "ABCDEFH" and not activite.startswith(("NRA", "NRT")):
        return activite[0]
    if activite in ("LULUCF", "OTHER") or activite.startswith(("NRA", "NRT")):
        return activite
    return "Other"


print(f"{matrice_defaut['Source_Code'].n_unique()} catégories, {len(ACTIVITES_OUTIL)} activités, gaz : {sorted(matrice_defaut['POLLUTANT'].unique())}")
matrice_defaut.filter(pl.col("POLLUTANT") == "CO2", pl.col("facteur") > 0, pl.col("Source_Code").is_in(["1.A.3.b", "1.A.4.a", "1.A.4.b", "1.A.4.c", "3.A", "5.A"]))
```

يجب أن تجد كل فئة من فئات الجرد سطرًا. وتحتاج رموز الطيران المحلي والملاحة المحلية إلى سطر أيضًا، حتى دون انبعاثات في الجرد، لأن الأداة تُدخل عبرها بيانات منظمة التعاون والتنمية في الميدان الاقتصادي عن نقل المقيمين (الصفحة 22). ودون سطر في المصفوفة، تضيع هذه البيانات.

```{python}
#| label: couverture
PREFIXES = correspondance["prefixe"].to_list()


def regle(code: str) -> str | None:
    """Préfixe le plus long de la correspondance qui couvre un code (« 3. » est couvert par « 3 »)."""
    code = code.rstrip(".")
    couvrants = [p for p in PREFIXES if code == p or code.startswith(p + ".")]
    return max(couvrants, key=len) if couvrants else None


# Les données de l'OCDE entrent par les codes de l'aviation et de la navigation intérieures
CODES_OCDE = [l[1] for l in plages["CRT"][1:] if l[11] in ("Domestic Aviation", "Domestic Navigation")]
codes = pl.concat([inventaire.select("code_outil"), pl.DataFrame({"code_outil": CODES_OCDE})]).unique().with_columns(
    prefixe=pl.col("code_outil").map_elements(regle, return_dtype=pl.String)
)
assert codes["prefixe"].null_count() == 0, f"Codes sans correspondance : {codes.filter(pl.col('prefixe').is_null())['code_outil'].to_list()}"
print(f"{codes.height} codes de l'outil (inventaire et OCDE : {CODES_OCDE}), tous couverts par {codes['prefixe'].n_unique()} lignes de la correspondance")
```

## المصفوفة التونسية

للمصفوفة التونسية سطر لكل سنة ورمز من رموز الأداة ونشاط. يساوي المعامل 1 في الإسناد المباشر، ويساوي مفتاح السنة في التوزيع. ومجموع معاملات الرمز الواحد يساوي 1. وتصلح المصفوفة لجميع الغازات، لأن القواعد لا تتوقف على الغاز.

```{python}
#| label: matrice-tunisienne
annees = pl.DataFrame({"annee": ANNEES}, schema={"annee": pl.Int16})
regles = codes.join(correspondance, on="prefixe")
directes = regles.filter(pl.col("methode") == "direct").join(annees, how="cross").with_columns(facteur=pl.lit(1.0))
reparties = regles.filter(pl.col("methode") == "cle").drop("activite").join(cles, on="cle")
COLONNES = ["annee", "code_outil", "activite", "facteur", "prefixe", "methode", "cle"]
matrice = pl.concat([directes.select(COLONNES), reparties.select(COLONNES)]).sort("annee", "code_outil", "activite")

sommes = matrice.group_by("annee", "code_outil").agg(pl.col("facteur").sum())
assert (sommes["facteur"] - 1).abs().max() < 1e-9, "Les facteurs d'un code doivent sommer à 1"
assert set(matrice["activite"]) <= set(activites["code"]) | {"LULUCF"}, "Activité inconnue"
print(f"{matrice.height} lignes, {matrice['code_outil'].n_unique()} codes, {matrice['activite'].n_unique()} activités")
matrice.filter(pl.col("annee") == 2022, pl.col("code_outil").str.starts_with("1.A.3"))
```

## الاختلافات عن المصفوفة الافتراضية

يقارن الجدول التالي، للفئات التي تُصدر انبعاثات في 2022، القسم الرئيسي من التصنيف الصناعي الدولي الموحد في المصفوفة الافتراضية بنظيره في المصفوفة التونسية. تغيّر الأسطر الموسومة بـ`oui` (نعم) في العمود `section_modifiee` النتيجة حتى على مستوى الأقسام. أما الأسطر الأخرى فلا تضيف سوى التفصيل حسب الفرع.

```{python}
#| label: differences
SECTION_SITE = dict(activites.select("code", "section_fmi").iter_rows()) | {"LULUCF": "LULUCF"}
emissions_2022 = (
    inventaire.filter(pl.col("annee") == 2022)
    .group_by("code_outil")
    .agg(pl.col("kt_eq_co2").sum())
    .filter(pl.col("kt_eq_co2").abs() > 0.05)
)
defaut = (
    matrice_defaut.filter(pl.col("facteur") > 0)
    .join(inventaire.filter(pl.col("annee") == 2022).select(POLLUTANT="gaz", Source_Code="code_outil").unique(), on=["POLLUTANT", "Source_Code"])
    .with_columns(section=pl.col("activite_outil").map_elements(section_outil, return_dtype=pl.String))
    .group_by("Source_Code", "section")
    .agg(pl.col("facteur").sum())
    .sort("facteur", descending=True)
    .group_by("Source_Code", maintain_order=True)
    .agg(section_defaut=pl.col("section").first(), activites_defaut=pl.col("section").unique().sort().str.join(", "))
    .rename({"Source_Code": "code_outil"})
)
tunisie = (
    matrice.filter(pl.col("annee") == 2022)
    .with_columns(section=pl.col("activite").replace_strict(SECTION_SITE))
    .group_by("code_outil", "prefixe")
    .agg(
        activites_tunisie=pl.when(pl.col("methode").first() == "cle")
        .then(pl.lit("clé ") + pl.col("cle").first())
        .otherwise(pl.col("activite").unique().sort().str.join(", ")),
        section_tunisie=pl.col("section").sort_by("facteur", descending=True).first(),
    )
)
differences = (
    emissions_2022.join(tunisie, on="code_outil", how="left")
    .join(defaut, on="code_outil", how="left")
    .with_columns(
        section_modifiee=pl.when(pl.col("section_defaut") != pl.col("section_tunisie")).then(pl.lit("oui"))
        .when(pl.col("section_defaut").is_null()).then(pl.lit("sans facteur"))
        .otherwise(pl.lit("non")),
        kt_eq_co2_2022=pl.col("kt_eq_co2").round(1),
    )
    .sort("code_outil")
    .select("code_outil", "kt_eq_co2_2022", "section_defaut", "activites_defaut", "section_tunisie", "activites_tunisie", "section_modifiee")
)
with pl.Config(tbl_rows=-1, fmt_str_lengths=60):
    print(differences.filter(pl.col("section_modifiee") != "non"))
print(differences.group_by("section_modifiee").agg(pl.len(), pl.col("kt_eq_co2_2022").sum().round(1)).sort("section_modifiee"))
```

## مسألة مفتوحة
يتعين التحقق من الاختيارات التي تبتعد عن المصفوفة الافتراضية مع المعهد الوطني للإحصاء والوكالة الوطنية للتحكم في الطاقة. وأهمها الاحتراق في التجارة والإدارات، المسند إلى الخدمات لا إلى البناء؛ والاحتراق السكني، المسند إلى الأسر؛ والنقل البري، الموزع بين الفروع والأسر بدلًا من إسناده إلى النقل؛ والانبعاثات المتسربة من نقل الغاز وتوزيعه، المسندة إلى فرع الكهرباء والغاز. وينفرد الجرد التونسي بنقطتين. تحمل الفئة 1.A.1.c.i (صنع الوقود الصلب) انبعاثات كبيرة، وتُسند إلى استخراج المحروقات مع بقية الفئة 1.A.1.c. وتحمل الفئة 1.A.4.b.ii (الآلات المتنقلة في القطاع السكني) كل احتراق الأسر، في كل سنة.
