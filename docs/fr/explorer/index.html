// Pills for one choice among a few (the year), as in a segmented control.
// Pastilles pour un choix parmi quelques options (l'année), comme un sélecteur segmenté.
function segmented(options, value, label) {
  const el = html`<div class="seg" role="radiogroup" aria-label=${label}></div>`;
  el.value = value;
  const buttons = options.map(o => {
    const b = html`<button type="button" role="radio">${yearFormat(o)}</button>`;
    b.addEventListener("click", () => { el.value = o; mark(); el.dispatchEvent(new Event("input", {bubbles: true})); });
    el.append(b);
    return [b, o];
  });
  const mark = () => buttons.forEach(([b, o]) => { b.classList.toggle("on", o === el.value); b.setAttribute("aria-checked", o === el.value); });
  mark();
  return el;
}

kpis = {
  const sum = (rows, y) => d3.sum(rows.filter(d => d.year === y), d => d.value);
  const prev = YEARS[YEARS.indexOf(year) - 1];
  const change = (rows, scale = 1) => {
    if (prev === undefined) return html`<span class="k-sub">${UI.first_year}</span>`;
    const c = sum(rows, year) / sum(rows, prev) - 1;
    return html`<span class="k-sub"><b class=${c >= 0 ? "up" : "down"}>${c >= 0 ? "▲" : "▼"} ${fmt(Math.abs(c) * 100)} %</b> ${UI.vs_prev}</span>`;

  const top = (rows) => {
    const t = d3.rollups(rows.filter(d => d.year === year), v => d3.sum(v, d => d.value), d => d.name).sort((a, b) => b[1] - a[1])[0];

  };
  const [user, userShare] = top(E), [emitter, emitterShare] = top(M);
  return html`<div class="ex-kpis">
    <div class="kpi"><span class="k-label">${UI.kpi_energy}, ${yearFormat(year)}</span>

// A table with a search box: rows are filtered on every column.
// Un tableau avec une zone de recherche : les lignes sont filtrées sur toutes les colonnes.
function searchTable(rows, columns, title) {
  const norm = (s) => (s ?? "").toString().normalize("NFD").replace(/[̀-ͯ]/g, "").toLowerCase();
  const input = html`<input type="search" placeholder=${UI.search} autocomplete="off">`;
  const count = html`<span class="t-count"></span>`;
  const body = html`<tbody></tbody>`;
  const cell = (c, r) => c.key === "section"
    ? html`<td><span class="pill" style=${"--c:" + (section_colors[r.section] ?? "#999")}>${r.section}</span></td>`
    : html`<td class=${c.cls ?? ""}>${r[c.key]}</td>`;
  const draw = () => {
    const q = norm(input.value);
    const shown = rows.filter(r => q === "" || columns.some(c => norm(r[c.key]).includes(q)));
    body.replaceChildren(...(shown.length ? shown.map(r => html`<tr>${columns.map(c => cell(c, r))}</tr>`)
      : [html`<tr><td colspan=${columns.length} class="empty">${UI.no_match}</td></tr>`]));
    count.textContent = `${shown.length} / ${rows.length} ${UI.rows}`;
  };
  input.addEventListener("input", (e) => { e.stopPropagation(); draw(); });
  draw();
  return html`<section class="ex-table">
    <header><h3>${title}</h3><label class="search">${input}</label>${count}</header>
    <div class="t-scroll"><table class="table table-sm">
      <thead><tr>${columns.map(c => html`<th>${c.label}</th>`)}</tr></thead>${body}</table></div>
  </section>`;
}
