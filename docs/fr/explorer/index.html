// Axes and bar order stay fixed across years, so that only real changes move.
// Les axes et l'ordre des barres restent fixes d'une année à l'autre.
totals = (rows) => d3.rollup(rows, v => d3.sum(v, d => d.value), d => d.year, d => d.name)
maxTotal = (rows) => d3.max(Array.from(totals(rows).values(), m => d3.max(m.values())))
stableOrder = (rows) => d3.groupSort(rows, v => -d3.sum(v, d => d.value), d => d.name)
